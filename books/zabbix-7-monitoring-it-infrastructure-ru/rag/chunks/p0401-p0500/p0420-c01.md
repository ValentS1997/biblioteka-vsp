    чтобы вы могли увидеть результат выполнения скрипта:

    # Запись результатов в файл
    def generate_host_file(hosts,host_file):
        hostname = None
        f = open(host_file, "w")
        # Добавьте записи об узлах сети, полученные из Zabbix
        for host in hosts:
            hostname = host['host']
            for interface in host["interfaces"]:
                if interface["main"] == "1":
                    f.write(hostname + " " + interface["ip"] + "\n")
        f.close()
        return
