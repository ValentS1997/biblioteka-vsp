package main
import "fmt"

func my_transform(in []int, xform func(int) int) (out []int) {
   out = make([]int, len(in))
   for idx, val := range in {
      out[idx] = xform(val)
   }
   return
}

func main() {
   data := []int{8, 6, 7, 5, 3, 0, 9}
   total := 0
   fmt.Printf("Result is %v\n", my_transform(data, func(v int) int {
                   total += v
                   return total
              }))
}
