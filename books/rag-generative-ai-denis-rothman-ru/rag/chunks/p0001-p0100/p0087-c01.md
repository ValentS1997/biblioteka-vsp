Создание механизма семантического поиска и генеративного агента     87

def fetch_and_clean(url):
    try:
        response = requests.get(url)
        response.raise_for_status()  # В случае провала запроса (например, 404)
                                     # вызываем исключение
        soup = BeautifulSoup(response.content, 'html.parser')

        # Отдаем приоритет "mw-parser-output"; в случае его отсутствия
        # используем класс "content"
        content = soup.find('div', {'class': 'mw-parser-output'})
            or soup.find('div', {'id': 'content'})
        if content is None:
            return None

        # Удаляем указанные разделы, включая вложенные
        for section_title in ['References', 'Bibliography', 'External links',
                              'See also', 'Notes']:
            section = content.find('span', id=section_title)
            while section:
                for sib in section.parent.find_next_siblings():
                    sib.decompose()
                section.parent.decompose()
                section = content.find('span', id=section_title)

        # Извлекаем и очищаем текст
        text = content.get_text(separator=' ', strip=True)
        text = clean_text(text)
        return text
    except requests.exceptions.RequestException as e:
        print(f"Error fetching content from {url}: {e}")
        return None    # При ошибке возвращаем None

В каждом проекте для исходных данных потребуются определенные имена
и пути. Здесь мы внесем дополнительную функцию, чтобы сохранять каждый
фрагмент текста с именем его источника, создавая ключевое слово на основе со
ответствующего URL-адреса:

# Каталог для хранения выходных файлов
output_dir = './data/'
os.makedirs(output_dir, exist_ok=True)

# Обработка каждого URL и запись его содержимого в отдельный файл
for url in urls:
    # Обработка расширения .html
    article_name = url.split('/')[-1].replace('.html',")
    # Создание имени файла для статьи
    filename = os.path.join(output_dir, article_name + '.txt')
    clean_article_text = fetch_and_clean(url)
    with open(filename, 'w', encoding='utf-8') as file:
        file.write(clean_article_text)

print(f"Content(ones that were possible) written to files
      in the '{output_dir}' directory.")
