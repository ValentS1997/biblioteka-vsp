Теперь отформатируем вывод с помощью textwrap и выведем результат. Здесь
сначала print_formatted_response(response) проверяет, содержит ли ответ
признаки Markdown. Если да, ответ форматируется. Если же нет, применяется
стандартное форматирование вывода по указанной ширине:

import textwrap
import re
from IPython.display import display, Markdown, HTML
import markdown

def print_formatted_response(response):
    # Проверка присутствия Markdown: заголовков, жирного шрифта, списков и пр.
    markdown_patterns = [
        r"^#+\s",           # Заголовки
        r"^\*+",            # Пункты списка
        r"\*\*",            # Жирный
        r"_",               # Курсив
        r"\[.+\]\(.+\)",    # Ссылки
        r"-\s",             # Тире в списках
        r"\`\`\`"           # Блоки кода
    ]

    # В случае обнаружения какого-либо паттерна предполагается,
    # что ответ представлен в формате Markdown
    if any(re.search(pattern, response, re.MULTILINE)
      for pattern in markdown_patterns):
        # Markdown обнаружен, преобразуем текст в HTML
        # для более качественного отображения
        html_output = markdown.markdown(response)
        display(HTML(html_output)) # Используем display(HTML())
                                   # для отображения HTML в Colab
