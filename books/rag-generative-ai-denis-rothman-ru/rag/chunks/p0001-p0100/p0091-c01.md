Добавим также код для отображения датасета. Начнем с загрузки данных
в DataFrame Pandas:

import json
import pandas as pd
import numpy as np

# Предполагается, что 'ds' – это ваш загруженный в Deep Lake датасет

# Создаем словарь для хранения данных
data = {}

# Перебираем тензоры датасета
for tensor_name in ds.tensors:
    tensor_data = ds[tensor_name].numpy()

    # Проверяем, является ли тензор многомерным
    if tensor_data.ndim > 1:
        # Преобразуем многомерные тензоры в одномерные
        data[tensor_name] = [np.array(e).flatten().tolist()
                             for e in tensor_data]
    else:
        # Преобразуем одномерные тензоры в списки и декодируем текст
        if tensor_name == "text":
            data[tensor_name] = [t.tobytes().decode('utf-8') if t else ""
                                 for t in tensor_data]
        else:
            data[tensor_name] = tensor_data.tolist()

# Создаем из словаря DataFrame Pandas
df = pd.DataFrame(data)
