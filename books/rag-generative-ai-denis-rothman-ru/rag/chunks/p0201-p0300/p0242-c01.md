import spacy
import numpy as np

# Загружаем обученную языковую модель spaCy
nlp = spacy.load('en_core_web_md')

Далее программа создает функцию оценки сходства, которая получает два ар
гумента, изначальный завершенный text1 и извлеченный text2, и возвращает
показатель их сходства:

def simple_text_similarity(text1, text2):
    # Преобразуем тексты в объекты документов spaCy
    doc1 = nlp(text1)
    doc2 = nlp(text2)

    # Получаем векторы для каждого документа
    vector1 = doc1.vector
    vector2 = doc2.vector

    # Вычисляем косинусное сходство между этими векторами
    # Проверяем наличие векторов, чтобы избежать деления на нуль
    if np.linalg.norm(vector1) == 0 or np.linalg.norm(vector2) == 0:
        return 0.0    # Возвращаем нуль, если один из текстов
                      # не имеет векторного представления
    else:
        similarity = np.dot(vector1, vector2) / (np.linalg.norm(vector1) *
          np.linalg.norm(vector2))
        return similarity
