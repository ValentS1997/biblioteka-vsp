246     Глава 8. Создание динамических RAG с помощью Chroma и Hugging Face Llama

Если обратиться к коллекции так, как мы делали в предыдущем разделе, исполь
зуя только один промт, то ответ вернется быстро:

import time
import textwrap

# Начинаем отсчет перед запросом
start_time = time.time()

# Обращаемся к коллекции, применяя промт
results = collection.query(
    # Используем промт в списке, как того ожидает метод выполнения запроса
    query_texts=[prompt],
    n_results=1    # Сколько результатов извлечь
)

# Измеряем время ответа
response_time = time.time() - start_time

# Выводим время ответа
print(f"Response Time: {response_time:.2f} seconds\n")

# Проверяем, были ли извлечены документы
if results['documents'] and len(results['documents'][0]) > 0:
    # Форматируем вывод для лучшей читаемости
    # Сокращаем длину строки вывода до 70 символов
    wrapped_question = textwrap.fill(prompt, width=70)
    wrapped_document = textwrap.fill(results['documents'][0][0], width=70)

    # Выводим отформатированные ответы
    print(f"Question: {wrapped_question}")
    print("\n")
    print(f"Retrieved document: {wrapped_document}")
    print()
else:
    print("No documents retrieved."
