Создание мультимодальной RAG для сферы БПЛА     123

def display_image_with_bboxes(image_data, bboxes, labels, label_name, ind=0):
    # Вывод изображения с рамками для конкретной метки

    image_bytes = io.BytesIO(image_data)
    img = Image.open(image_bytes)

    # Извлечение имен классов для выбранного изображения
    class_names = ds.labels[ind].info['class_names']

    # Вывод на основе конкретной метки
    # (или вывод всего, если имена классов отсутствуют)
    if class_names is not None:
        try:
            label_index = class_names.index(label_name)
            relevant_indices = np.where(labels == label_index)[0]
        except ValueError:
            print(f"Warning: Label '{label_name}' not found. Displaying all boxes.")
            relevant_indices = range(len(labels))
    else:
        relevant_indices = []   # Метки не обнаружены, значит, рамки не выводим

    # Рисуем рамки
    draw = ImageDraw.Draw(img)
    for idx, box in enumerate(bboxes):   # Перечисляем bboxes
        if idx in relevant_indices:   # Проверяем соответствие рамки
            x1, y1, w, h = box
            x2, y2 = x1 + w, y1 + h
            draw.rectangle([x1, y1, x2, y2], outline="red", width=2)
            draw.text((x1, y1), label_name, fill="red")
    # Сохраняем изображение
    save_path="boxed_image.jpg"
    img.save(save_path)
    display(img)
