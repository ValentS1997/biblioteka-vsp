Теперь создадим функцию, которая будет вычислять размер пакетов, ограничи
вая его до 4 Мбайт. Это примерно равно текущему лимиту Pinecone:

# Функция для вычисления размера пакета
# На всякий случай лимит установлен чуть ниже 4 Мбайт
def get_batch_size(data, limit=4000000):
    total_size = 0
    batch_size = 0
    for item in data:
        item_size = sum([sys.getsizeof(v) for v in item.values()])
        if total_size + item_size > limit:
            break
        total_size += item_size
        batch_size += 1
    return batch_size
