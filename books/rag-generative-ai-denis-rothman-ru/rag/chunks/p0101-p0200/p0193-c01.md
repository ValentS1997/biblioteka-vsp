Запрос преобразован в вектор. Теперь извлечем k профилей клиентов, макси
мально схожих с целевым вектором, и спарсим результат:

# Выполняем запрос, используя созданный эмбеддинг
query_results = index.query(
    vector=query_embedding,
    top_k=5,
    include_metadata=True,
)

Выводим ответ и сопутствующие метаданные:
# Выводим результат и метаданные
print("Query Results:")
for match in query_results['matches']:
    print(f"ID: {match['id']}, Score:  {match['score']}")
    if 'metadata' in match and 'text' in match['metadata']:
        print(f"Text: {match['metadata']['text']}")
    else:
        print("No metadata available.")

response_time = time.time() - start_time    # Измеряем время ответа
# Выводим время ответа
print(f"Querying response time: {response_time:.2f} seconds")
