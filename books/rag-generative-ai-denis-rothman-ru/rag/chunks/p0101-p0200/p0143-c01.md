if matched_keyword:
    print(f"Fetching data from: {urls[matched_keyword]}")
    cleaned_text = fetch_and_clean(urls[matched_keyword])

    # Ограничиваем вывод до указанного числа ключевых слов из очищенного текста
    words = cleaned_text.split()    # Разделяем текст на слова
    # Объединяем первые n слов в одну строку
    first_n_words = ' '.join(words[:num_words])

Параметр num_words служит для разбивки текста на чанки. И хотя такой подход
может работать при небольших объемах данных, в более сложных сценариях
рекомендуется преобразовывать данные в векторы.

Далее очищенный и обрезанный текст форматируется для вывода:
    # Ограничиваем ширину вывода первых n слов до 80 символов
    wrapped_text = textwrap.fill(first_n_words, width=80)
    print("\nFirst {} words of the cleaned text:".format(num_words))
    print(wrapped_text)   # Выводим первые n слов в отформатированном абзаце

    # В целях сохранения согласованности при составлении промта
    # для модели GPT-4o используем те же first_n_words
    prompt = f"Summarize the following information
      about {matched_keyword}:\ n{first_n_words}"
    wrapped_prompt = textwrap.fill(prompt, width=80)    # Перенос текста промта
    print("\nPrompt for Generator:", wrapped_prompt)

    # Возвращаем указанное число слов
    return first_n_words

Обратите внимание, что эта функция в итоге возвращает первые n слов, предо
ставляя лаконичный и релевантный фрагмент информации на основе запроса
пользователя. Такой дизайн позволяет системе эффективно управлять извлече
нием данных с сохранением участия пользователя.
