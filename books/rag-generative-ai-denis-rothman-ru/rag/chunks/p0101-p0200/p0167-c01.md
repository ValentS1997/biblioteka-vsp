Компонент 1: сбор и подготовка данных     167

Далее программа считывает этот файл JSON и устанавливает значения перемен
ных среды для аутентификации на Kaggle, задействуя ваше имя пользователя
и ключ API:

import os
import json

with open(os.path.expanduser("drive/MyDrive/files/kaggle.json"), "r") as f:
  kaggle_credentials = json.load(f)
kaggle_username = kaggle_credentials["username"]
kaggle_key = kaggle_credentials["key"]

os.environ["KAGGLE_USERNAME"] = kaggle_username
os.environ["KAGGLE_KEY"] = kaggle_key
