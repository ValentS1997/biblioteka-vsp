338                                             Часть III. 64-битовые режимы

 Листинг 20.6. Вложенный вызов 64-битовой подпрограммы (cg64_2.a)

%define delay   1000000                                        ; (8)
        use64                                                 ; (12)
        align   8                                             ; (13)

addr80  dq     0                                             ; (14)
        dw      cg1 - gdt                                      ; (15)

iter    db     8                                             ; (16)

subr_1:                                                       ; (17)
.l1:    _brk                                                  ; (18)
%ifdef _SHOW_                                                 ; (19)
        mov     al, [iter]                                     ; (20)
        add     al, '0'                                        ; (21)
        mov     [edi + 0xb8000], al                             ; (22)

        mov     rcx, delay                                     ; (23)
.l3:    jmp     .l4                                           ; (24)
.l4:    loop    .l3                                           ; (25)
%endif                                                       ; (26)
        cmp     byte [iter], 0                                 ; (27)
        ja      .l2                                           ; (28)
        retfq                                                 ; (29)
.l2:                                                         ; (30)
        dec     byte [iter]                                    ; (31)
        add     edi, 2                                         ; (32)
        call    far qword [addr80]                              ; (33)
        sub     edi, 2                                         ; (34)
        jmp     .l1                                           ; (35)

long_mode:                                                    ; (127)
.l2:    _brk                                                  ; (128)
%ifdef _SHOW_                                                 ; (129)
        store   0xb8002, byte [1]                              ; (130)
        inc     byte [1]                                       ; (131)
%endif                                                       ; (132)
        mov     edi, 10                                        ; (133)
        call    (cg1 - gdt):0      ; !                         ; (134)
        jmp     .l2                                           ; (135)
end:                                                         ; (136)
