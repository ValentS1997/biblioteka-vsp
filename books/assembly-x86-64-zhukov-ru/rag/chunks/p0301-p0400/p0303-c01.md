Глава 17. Преобразование адресов                                              303

рис. 17.6 показаны две верхние строки редактора, с адреса 0x37100 начинается код
программы, что можно проверить по lst-файлу. Число (17), выделенное на рис. 17.6
полужирным шрифтом, отмечает конец com-файла.

Чтобы запустить этот код, нужно сделать программу, которая настроит преобразо-
вание адресов 0 ->  37000, обнулит базовые линейные адреса данных и стека и
передаст управление по линейному адресу 0x100, при этом также обнулив базовый
линейный адрес кода. Программа приведена в листинге 17.13.

 37100   DB8766FC  123A166 3D8B0000      125 8000C781 F0B9000B F3000000 8766AB66
 37120   2AFBEBDB    78007       0 55555500  7290755  7310720  7390739  72D0731

                     Рис. 17.6. Программа test.com в памяти с адреса 0x37100

 Листинг 17.13. Программа для вызова модуля test.com (pg7.asm)

%define tsr_seg 0x3632                                         ; (3)
%define dir 0                                                 ; (4)
%define pt0 (dir + 0x1000)                                     ; (5)
dsc8:                                                        ; (116)
flat_code:                    ; Code (flat)                   ; (117)
        dw      -1             ; limit low (max)                ; (118)
        dw     0              ; base low                      ; (119)
        db     0              ; base middle                   ; (120)
        db      1001_1000b     ; P, dpl=00, S  | Code, c, xr, a  ; (121)
        db      1100_1111b     ; G, 32 bit, 00 | limit high (max); (122)
        db     0                                             ; (123)
        bzero   dir, 0x2000                                    ; (179)

        mov     edi, pt0                                       ; (180)
        mov     eax, 0 + pg_flags   ; phadr := 0                 ; (181)
        mov     ecx, 1024                                      ; (182)
.l1:                                                         ; (183)
        stosd                     ; pt0[i] := phadr            ; (184)
        add     eax, 0x1000        ; phadr + 4k                 ; (185)
        loop    .l1                                           ; (186)

        mov     eax, pt0 + pg_flags                             ; (187)
        mov     edi, dir                                       ; (188)
        stosd                     ; dir[0] := phadr(pt0)       ; (189)

        tsr_pg  equ 1 + ((tsr_seg + 1) >> 8)                    ; (190)
        tsr_adr equ tsr_pg * 0x1000                             ; (191)
