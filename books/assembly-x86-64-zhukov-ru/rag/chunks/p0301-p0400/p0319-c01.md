 Листинг 18.9. Ограничения доступа к страницам (pg_1.asm)

%define pg_present     1                                      ; (5)
%define pg_user_rw     2                                      ; (6)
%define pg_user_level   4                                      ; (7)
%define pg_work (pg_present + pg_user_rw + pg_user_level)        ; (8)
%define pg_test (pg_present + pg_user_level)                    ; (9)
proc_3:                                                       ; (14)
.l1:                                                         ; (15)
        _brk                                                  ; (16)
%ifdef _SHOW_                                                 ; (17)
        es mov  [0xb8000 + 3*2], al                             ; (18)
%endif                                                       ; (19)
        inc     al                                            ; (20)
        es mov  ah, [0]        ; read page 0                   ; (21)
        es mov  [0], al        ; write page 0                  ; (22)
        jmp     .l1                                           ; (23)
        mov     edi, pt0                                       ; (104)
        mov     eax, 0 + pg_test    ; !                         ; (105)
        stosd                                                 ; (106)

        mov     eax, 0 + pg_work                               ; (107)
        mov     ecx, 1023                                      ; (108)
.l1:                                                         ; (109)
        add     eax, 4096          ; phadr + 4k                 ; (110)
        stosd                     ; pt0[i] := phadr            ; (111)
        loop    .l1                                           ; (112)

        mov     eax, pt0 + pg_work                              ; (113)
        mov     edi, dir                                       ; (114)
        stosd                     ; dir[0] := phadr(pt0)       ; (115)
        mov     ecx, 1023          ; dir[1..1023] := 0          ; (116)
        clr     eax                                           ; (117)
        rep     stosd                                          ; (118)

В строках (5–7) определены биты PDE, в (8) — их объединение, которое открывает
уровню user доступ на чтение-запись. Оно используется в (113) и (107) при инициа-
лизации каталога и таблицы страниц с номерами 1–1023. В строке (9) задано час-
тичное объединение, которое разрешает доступ только на чтение, оно используется
в (105) при инициализации нулевой страницы памяти.
