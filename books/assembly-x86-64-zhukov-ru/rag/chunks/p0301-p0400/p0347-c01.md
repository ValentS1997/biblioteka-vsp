 Листинг 21.4. Перемещаемая 64-битовая программа (rel64.a)

%define reloc_ofs 0x100                                        ; (8)
        use32                                                 ; (61)
pm:                                                          ; (62)
reloc_sz equ    end64 - start64                                ; (71)

reloc_64_bit_code:                                            ; (72)
        std                                                   ; (73)
        mov     esi, start64 + reloc_sz - 1                     ; (74)
        lea     edi, [esi + reloc_ofs]                          ; (75)
        mov     ecx, reloc_sz                                  ; (76)
        rep     movsb                                          ; (77)
        mov     al, 0x55                                       ; (78)
        mov     ecx, reloc_ofs                                 ; (79)
        rep     stosb                                          ; (80)
        cld                                                   ; (81)
        use64                                                 ; (117)
        default rel                                           ; (118)

        align   0x100                                          ; (119)
start64:                                                      ; (120)

vendor  resd    3                                             ; (121)
str8    db      '00000000'                                     ; (154)
end64:                                                       ; (155)
