 Листинг 7.5. Рабочий вариант решения задачи (var_2.8)

dos     macro                   ; (1)
        mov     ah, #1           ; (2)
        int     021              ; (3)
#em                             ; (4)

        mov     bx, 0            ; (5)
        mov     cx, 08000        ; (6)
        lea     dx, buf          ; (7)
        dos     03f              ; (8)
        jc      >l9              ; (9)

        mov     cx, ax           ; (10)
        jcxz    >l9              ; (11)

        call    make             ; (12)

        mov     bx, 1            ; (13)
        lea     dx, buf          ; (14)
        dos     040              ; (15)
        jc      >l9              ; (16)

        mov     al, 0            ; (17)
        dos     04c              ; (18)
l9:                             ; (19)
        mov     al, 1            ; (20)
        dos     04c              ; (21)

make:                           ; (22)
        push    cx               ; (23)
        lea     si, buf          ; (24)
        mov     di, si           ; (25)
        cld                     ; (26)
l1:                             ; (27)
        lodsb                   ; (28)
        mov     ah, 0            ; (29)
        clc                     ; (30)
l2:                             ; (31)
        adc     ah, 0            ; (32)
        shr     al, 1            ; (33)
        jnz     l2               ; (34)

        mov     al, ah           ; (35)
        adc     al, 0            ; (36)
        stosb                   ; (37)
        loop    l1               ; (38)
