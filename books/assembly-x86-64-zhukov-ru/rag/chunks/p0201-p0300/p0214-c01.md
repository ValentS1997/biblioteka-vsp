214                                                 Часть I. Реальный режим

В листинге 12.10 приведены примеры перечисленных команд.

 Листинг 12.10. Команды, появившиеся в 32-разрядных процессорах (new_cmd.8x)

        mov     al, -2                         ; (1)
        cbw                   ; al -> ax      ; (2)
        cwd                   ; ax -> dx:ax    ; (3)
        cwde                  ; ax -> eax     ; (4)

        mov     eax, 080                       ; (5)
        movsx   bx, al                         ; (6)

        mov     eax, 0aa550000                  ; (7)
        movzx   eax, bx                        ; (8)

        pushd   1, 2, 3, 4, 5, 6, 7, 8          ; (9)
        popad                                 ; (10)

        mov     esi, bit 30                    ; (11)
        mov     ecx, 32                        ; (12)
l1:     bt      esi, ecx                       ; (13)
        jc      >l2                            ; (14)
        loop    l1                             ; (15)
l2:                                           ; (16)
        stc                                   ; (17)
        setc    ch                             ; (18)
        cmovc   bx, bp                         ; (19)
        clc                                   ; (20)
        setc    ch                             ; (21)
        cmovc   bx, dx                         ; (22)

        xadd    eax, esi                       ; (23)

        mov     eax, 0                         ; (24)
        bsf     dx, eax                        ; (25)
        mov     eax, bit 16 + bit 15            ; (26)
        bsf     dx, eax                        ; (27)
        bsr     dx, eax                        ; (28)

        mov     ax, 02076                      ; (29)
        mov     dx, 3                          ; (30)
        shrd    ax, dx, 3                      ; (31)

        mov     ax, 02076                      ; (32)
        mov     dx, 0c000                      ; (33)
        shld    ax, dx, 2                      ; (34)

        int     020                            ; (35)
