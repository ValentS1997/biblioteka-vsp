254                                              Часть II. Защищенный режим

(код 1111). В чем разница? Только в том, что при вызове процедуры через Interrupt
Gate автоматически сбрасывается флаг разрешения прерываний (i), а через Trap
Gate — не сбрасывается2.

В листинге 15.1 приведен пример обработки исключений с номерами ноль и один.
В позициях экрана с этими же номерами отображаются счетчики исключений.

 Листинг 15.1. Обработка исключений 0 и 1 (isr_2.asm)

        ..
; ------ isr                                                  ; (13)

i_cnt:  resd    2                                             ; (14)

isr0:   push    edx                                           ; (15)
        clr     edx                                           ; (16)
        jmp     isr                                           ; (17)

isr1:   push    edx                                           ; (18)
        mov     edx, 1                                         ; (19)
        jmp     isr                                           ; (20)

; to be continued...                                           ; (21)

isr:    _brk                                                  ; (22)
        push    eax                                           ; (23)
        lea     eax, [i_cnt + (edx * 4)]                        ; (24)
        inc     dword [eax]                                    ; (25)

%ifdef _SHOW_                                                 ; (26)
        mov     eax, [eax]                                     ; (27)
        call    wr_hex                                         ; (28)
%endif                                                       ; (29)
        pop     eax                                           ; (30)
        pop     edx                                           ; (31)
        iret                                                  ; (32)

;------- IDT                                                  ; (90)
idt:                                                         ; (91)
i0:                                                          ; (92)
        dw      isr0           ; low adr                       ; (93)
        dw      code - gdt     ; selector                      ; (94)
        db     0              ; (rsrv)                        ; (95)
        db      1000_1111b     ; P, dpl, 0 | Trap               ; (96)
        dw     0              ; high adr                      ; (97)
