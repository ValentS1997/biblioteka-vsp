224                                              Часть II. Защищенный режим

        align   4              ; !                              ; (4)
gdt:                                                          ; (5)
dsc0:                         ; Null                           ; (6)
        dd      0, 0                                           ; (7)
dsc1:                         ; Code 32 bit                    ; (8)
        dw      end - 1        ; limit low                    ! ; (9)
        dw     0              ; base low                     ! ; (10)
        db     0              ; base middle                  ! ; (11)
        db      1001_1000b     ; P, dpl, S | 1, c, xr, a       ! ; (12)
        db      0100_0000b     ; g, 32 bit, 00 | limit high     ! ; (13)
        db     0              ; base high                      ; (14)
gdtr:                                                         ; (15)
        dw     $ - gdt - 1                                     ; (16)
        dd      gdt            ; +                              ; (17)

;---------------------------                                    ; (18)
start:  _brk                                                   ; (19)

        ladr_cs                ; +                              ; (20)
        mov     edx, eax                                        ; (21)

        wr_base dsc1           ; +                              ; (22)
        add     [gdtr+2], edx   ; !                              ; (23)
        lgdt    [gdtr]                                          ; (24)

        cli                                                    ; (25)
        switch_mode            ; +                              ; (26)
        jmp     8:pm           ; +                              ; (27)

;---------------------------                                    ; (28)
;       org     0200h          ; +                              ; (29)
        use32                                                  ; (30)

pm:     _brk                                                   ; (31)
        clr     ebx, ecx                                        ; (32)
l2:     inc     word [ebx]                                      ; (33)
;       cs mov  al, [dsc1+5]    ; ?                              ; (34)
;       waste   4*1024         ; ?                              ; (35)
        dec     ecx                                            ; (36)
        jmp     l2                                             ; (37)
;---------------------------                                    ; (38)
end:                          ; !                              ; (39)

Отличия от листинга 13.1 помечены знаками '!', '+' и '?'. Восклицанием отмече-
ны изменения, вынужденные различиями между a386 и nasm, плюсом — усовер-
шенствования, а вопросом — команды, которые мы пустим в ход при последующих
опытах.
