236                                              Часть II. Защищенный режим

dsc5:                         ; Data (flat), r/w                ; (46)
        dw      -1             ; limit low (max)                 ; (47)
        dw     0              ; base low                       ; (48)
        db     0              ; base middle                    ; (49)
        db      1001_0010b     ; P, dpl, S | data, e, RW, a      ; (50)
        db      1100_1111b     ; G, B, 00 | limit high (max)     ; (51)
        db     0                                              ; (52)

        ladr_cs                ; eax := ladr(cs)                 ; (59)
        mov     edx, eax                                        ; (60)

        wr_base dsc1                                           ; (61)

        mov     eax, edx                                        ; (62)
        wr_base dsc2                                           ; (63)

        mov     eax, edx                                        ; (64)
        wr_base dsc4                                           ; (65)

        jmp     (dsc1 - gdt):pm                                 ; (70)

        use32                                                  ; (72)
pm:     _brk                                                   ; (73)

        set_seg ds, dsc4 - gdt                                  ; (74)
        set_seg es, dsc5 - gdt                                  ; (75)

        mov     esi, 080h          ; cmd string                  ; (76)
        mov     edi, 0b8000h                                    ; (77)
        clr     ecx                                            ; (78)
        cld                                                    ; (79)
        lodsb                                                  ; (80)
        mov     cl, al                                          ; (81)
        jcxz    l2                                             ; (82)

l1:     movsb                                                  ; (83)
        inc     edi                                            ; (84)
        loop    l1                                             ; (85)

l2:     not     word [smth]                                     ; (86)
        call    (cg1 - gdt):subr    ; call gate                  ; (87)
        jmp     l2                                             ; (88)
end:                                                          ; (89)
