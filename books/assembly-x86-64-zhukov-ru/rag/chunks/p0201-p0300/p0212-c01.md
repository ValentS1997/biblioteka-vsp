 Листинг 12.9. Поиск в массиве с использованием масштабируемого индекса
 (find_3.8x)

x       equ     012345678                        ; (1)
max32   equ     07fffffff      ; signed          ; (2)
min32   equ     080000000                        ; (3)

        jmp     start                            ; (4)

array   dd      0, max32, min32, x-1, x+1, x      ; (5)
item_sz equ     type array                       ; (6)
item_no equ     ($ - array) / item_sz             ; (7)

start:  mov     cx, item_no                      ; (8)
        mov     esi, -1                          ; (9)
l1:                                            ; (10)
        inc     esi                              ; (11)
        mov     eax, [esi*item_sz+array]          ; (12)
        cmp     eax, x                           ; (13)
        loopne  l1                              ; (14)
        jne     miss                             ; (15)

found:  int     3                               ; (16)
miss:   int     020                              ; (17)
