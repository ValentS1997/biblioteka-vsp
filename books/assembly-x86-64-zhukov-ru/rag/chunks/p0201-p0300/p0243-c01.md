Глава 14. Данные и стек                                                       243

 Листинг 14.7. Проверка макрокоманд управления линией A20 (a20.asm)

dsc2:                         ; Data (flat), r/w                ; (15)
        dw      -1             ; limit low (max)                 ; (16)
        dw     0              ; base low                       ; (17)
        db     0              ; base middle                    ; (18)
        db      1001_0010b     ; P, dpl, S | data, e, RW, a      ; (19)
        db      1100_1111b     ; G, B, 00 | limit high (max)     ; (20)
        db     0                                              ; (21)

        switch_mode                                            ; (33)
        jmp     8:pm                                           ; (34)

        use32                                                  ; (36)

write_compare:                                                 ; (37)
        mov     ebx, 0x100000                                   ; (38)
        mov     edx, 0x10                                       ; (39)
        cld                                                    ; (40)
        mov     edi, ebx                                        ; (41)
        mov     ecx, edx                                        ; (42)
l1:                                                           ; (43)
        mov     eax, edi                                        ; (44)
        stosb                                                  ; (45)
        loop    l1                                             ; (46)

        clr     esi                                            ; (47)
        mov     edi, ebx                                        ; (48)
        mov     ecx, edx                                        ; (49)
        repe    cmpsb                                           ; (50)
        sete    al     ; al = 1 if identical                    ; (51)
        ret                                                    ; (52)

pm:     _brk                                                   ; (53)
        set_seg ds, dsc2 - gdt                                  ; (54)
        set_seg es, dsc2 - gdt                                  ; (55)

        a20_on                                                 ; (56)
        call    write_compare   ; al = 0 ?                       ; (57)
        _brk                                                   ; (58)

        a20_off                                                ; (59)
        call    write_compare   ; al = 1 ?                       ; (60)
l2:     _brk                                                   ; (61)
        jmp     l2                                             ; (62)
