270                                              Часть II. Защищенный режим

В листинге 16.1 приведен простейший пример, где в LDT задан только один деск-
риптор2.

 Листинг 16.1. Простой пример LDT с одним дескриптором (ldt.asm)

gdt:                                                         ; (84)
dsc0:                         ; Null                          ; (85)
        dd      0, 0                                           ; (86)
dsc7:                         ; !                             ; (134)
lt1:                          ; Data, r/w                     ; (135)
        dw      ldt_sz         ; limit low                     ; (136)
        dw     0              ; base low                      ; (137)
        db     0              ; base middle                   ; (138)
        db      1000_0010b     ; P, dpl=00, s | ldt             ; (139)
        db     0              ; flags + limit high             ; (140)
        db     0                                             ; (141)

gdtr:                                                        ; (142)
        dw     $ - gdt - 1     ; size(gdt)                     ; (143)
        dd      gdt            ; + ladr(cs)                    ; (144)

;------- LDT                                                  ; (145)
ldt:                          ; !                             ; (146)
l_dsc0:                       ; Data (video), r/w              ; (147)
        dw      160            ; limit low                     ; (148)
        dw      0x8000         ; base low (0xb8000)             ; (149)
        db      0xb            ; base middle                   ; (150)
        db      1001_0010b     ; P, dpl=00, S  | Data, e, RW, a  ; (151)
        db      0100_0000b     ; g, 32 bit, 00 | limit high (0)  ; (152)
        db     0                                             ; (153)

ldt_sz  equ     $ - ldt                                        ; (154)
;---------------------------                                   ; (189)
start:                                                       ; (190)
        mov     eax, edx                                       ; (204)
        add     eax, ldt                                       ; (205)
        wr_base lt1            ; !                             ; (206)

        add     [gdtr+2], edx                                  ; (207)
        add     [idtr+2], edx                                  ; (208)

        lgdt    [gdtr]                                         ; (209)
        lidt    [idtr]                                         ; (210)
