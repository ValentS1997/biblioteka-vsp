282                                              Часть II. Защищенный режим

Хотя программа hw2.com в Bochs работает нормально, при выполнении на PC она
сразу приводит к перезагрузке системы. Чтобы выяснить причину сбоя, в hw2.asm
пришлось добавить обработку fault-исключений (примерно так, как это сделано
в программе isr_n.asm из листинга 15.2) с выводом счетчиков и дополнительно кода
ошибки, а также отображением содержимого TSS (листинг 16.6).

 Листинг 16.6. Программа для диагностики сбоя при использовании TSS (err2.asm)

%define trap_no    32                                          ; (3)
%define trap_limit 100                                         ; (4)
%ifdef _SHOW_                                                 ; (11)
view_tss:                         ; esi = ptr(ts), edx - pos    ; (12)
        cld                                                   ; (13)
        mov     ecx, tss_size / 4                              ; (14)
.l1:                                                         ; (15)
        lodsd                                                 ; (16)
        call    wr_hex                                         ; (17)
        inc     edx                                           ; (18)
        loop    .l1                                           ; (19)
        ret                                                   ; (20)
%endif                                                       ; (21)
; ------ trap                                                 ; (25)
%macro  set_trap 1                                            ; (26)
trap%1:  _brk                                                 ; (27)
        enter   0, 0                                           ; (28)
        push    edx                                           ; (29)
        mov     edx, %1 * 2                                    ; (30)
        jmp     trap                                           ; (31)
%endmacro                                                     ; (32)
trap:                                                        ; (44)
        push    eax                                           ; (45)
        lea     eax, [i_cnt + (edx * 4)]                        ; (46)
        inc     dword [eax]                                    ; (47)
        mov     eax, [eax]                                     ; (48)
.l2:    _brk                                                  ; (49)
        cmp     al, trap_limit                                 ; (50)
        ja      .l2                                           ; (51)

%ifdef _SHOW_                                                 ; (52)
        call    wr_hex                                         ; (53)
        mov     eax, [ebp+4]    ; err_code                      ; (54)
        inc     edx                                           ; (55)
