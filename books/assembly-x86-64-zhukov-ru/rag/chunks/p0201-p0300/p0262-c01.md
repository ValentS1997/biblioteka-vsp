262                                              Часть II. Защищенный режим

 Листинг 15.5. Обработка прерываний с перенастройкой контроллера (tm3.asm)

%define trap_no pic_vect                                       ; (3)
%define intr_no 8                                             ; (4)

; ------ isr                                                  ; (15)
        align   4                                             ; (16)
i_cnt:  resd    trap_no + intr_no                              ; (17)

%macro  set_trap 1                                            ; (18)
trap%1: _brk                                                  ; (19)
        push    edx                                           ; (20)
        mov     edx, %1                                        ; (21)
        jmp     exit                                           ; (22)
%endmacro                                                     ; (23)

%macro  set_traps 0                                           ; (24)
%assign i 0                                                   ; (25)
%rep trap_no                                                  ; (26)
        set_trap i                                            ; (27)
        %assign i i+1                                          ; (28)
%endrep                                                       ; (29)
%endmacro                                                     ; (30)

%macro  set_intr 1                                            ; (31)
intr%1: _brk                                                  ; (32)
        push    edx                                           ; (33)
        mov     edx, %1 + trap_no                              ; (34)
        jmp     irq%1                                          ; (35)
%endmacro                                                     ; (36)

%macro  set_intrs 0                                           ; (37)
%assign i 0                                                   ; (38)
%rep intr_no                                                  ; (39)
        set_intr i                                            ; (40)
        %assign i i+1                                          ; (41)
%endrep                                                       ; (42)
%endmacro                                                     ; (43)

        set_traps                                             ; (44)
        set_intrs                                             ; (45)

irq0:                                                        ; (46)
irq2:                                                        ; (47)
irq3:                                                        ; (48)
