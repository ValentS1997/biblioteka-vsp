8.5. Пример реализации REST API веб-сервиса      449

  Кроме того, можно задать, какой тип HTTP-сообщения будет обрабатывать
связанная с URL функция:

   func main() {
       // http://127.0.0.1:8080/api/v1/todo
       // http://127.0.0.1:8080/api/v1/todo/project/10/task/2
       // http://127.0.0.1:8080/api/v1/todo/project?projID=10&name=Oo
       router := mux.NewRouter().StrictSlash(true)
       router.HandleFunc("/api/v1/todo", home)
       router.HandleFunc("/api/v1/todo/project/{projID}/task/{taskID}",
                          ValuesInPath).Methods("GET")
       router.HandleFunc("/api/v1/todo/project", QueryString).Methods("GET", "POST")
       fmt.Println("Listening at port", PORT)
       log.Fatal(http.ListenAndServe(":"+PORT, router))
   }
  А проверить, какой тип HTTP-сообщения пришел на обработку, можно с по
мощью поля Method экземпляра типа http.Request:

   func QueryString(w http.ResponseWriter, r *http.Request) {
       switch r.Method {
       case "GET":
           fmt.Fprintf(w, "It is GET!!! ")
       case "POST":
           fmt.Fprintf(w, "It is POST!!! ")
       }
       queryStr := r.URL.Query() // получение строки запроса
       fmt.Printf("%+v\n", queryStr)

       answer := fmt.Sprintf("%+v", queryStr)
       fmt.Fprintf(w, answer)
   }
  Однако вернемся к разработке сервиса. В каталог service
добавьте эти файлы (рис. 8.22).
  В файл logger.go добавим код, который будет выполнять
логирование запросов, поступающих от клиента:

   // baseURL/part_8/8.5/golang/todo-service/service/logger.go
   package service
                                                                              Рис. 8.22. Структура
   import (                                                                           каталога service       "log"
       "net/http"
       "time"
   )

   func Logger(inner http.Handler, name string) http.Handler {
       return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
           start := time.Now()

           inner.ServeHTTP(w, r)

           log.Printf(
               "%s %s %s %s",
               r.Method,
               r.RequestURI,
               name,
               time.Since(start),
           )
       })
   }
