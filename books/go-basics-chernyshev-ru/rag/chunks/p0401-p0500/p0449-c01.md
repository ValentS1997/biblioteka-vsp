448       Глава 8   Сетевое программирование

  Если в конец строки запроса добавить сим
вол /, это приведет к тому, что страница не будет
найдена (рис. 8.21).
  Во избежание таких ошибок создайте экзем
пляр маршрутизатора, используя эту строчку
кода:
   router := mux.NewRouter().StrictSlash(true)       Рис. 8.21. Неправильное обращение к сервису
   Теперь посмотрим, как извлечь данные из      по пути http://127.0.0.1:8080/api/v1/todo/
пути или строки запроса:

   package main

   import (
       "fmt"
       "log"
       "net/http"

       "github.com/gorilla/mux"
   )

   const PORT = "8080"

   func home(w http.ResponseWriter, r *http.Request) {
       fmt.Fprintf(w, "Hello Wold in REST API style!!!")
   }

   func ValuesInPath(w http.ResponseWriter, r *http.Request) {
       params := mux.Vars(r) // получение параметров из пути
       fmt.Printf("%+v\n", params)

       answer := fmt.Sprintf("%+v", params)
       fmt.Fprintf(w, answer)
   }

   func QueryString(w http.ResponseWriter, r *http.Request) {
       queryStr := r.URL.Query() // получение строки запроса
       fmt.Printf("%+v\n", queryStr)

       answer := fmt.Sprintf("%+v", queryStr)
       fmt.Fprintf(w, answer)
   }

   func main() {
       // http://127.0.0.1:8080/api/v1/todo
       // http://127.0.0.1:8080/api/v1/todo/project/10/task/2
       // http://127.0.0.1:8080/api/v1/todo/project?projID=10&name=Oo
       router := mux.NewRouter().StrictSlash(true)
       router.HandleFunc("/api/v1/todo", home)
       router.HandleFunc("/api/v1/todo/project/{projID}/task/{taskID}", ValuesInPath)
       router.HandleFunc("/api/v1/todo/project", QueryString)
       fmt.Println("Listening at port", PORT)
       log.Fatal(http.ListenAndServe(":"+PORT, router))
   }
   // при обращении через браузер по пути
   // http://127.0.0.1:8080/api/v1/todo/project/10/task/2
   // map[projID:10 taskID:2]
   // http://127.0.0.1:8080/api/v1/todo/project?projID=10&name=Oo
   // map[name:[Oo] projID:[10]]
