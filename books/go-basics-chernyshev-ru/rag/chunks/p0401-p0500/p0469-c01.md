изменений, — board. Откройте файл board.go и добавьте в структуру Board метод

   func (b *Board) IsEmpty() bool {
       for i := range b.Size {
           for j := range b.Size {
               if b.Board[i][j] ! = Empty {
                   return false
               }
           }

  Следующее изменение внесем в файл board_cell_type.go, добавив переопре
деление метода String() для типа BoardField, что позволит не писать лишний раз
проверки того, какой символ выводить при его передаче на вход форматированной
функции fmt.Printf:

   // baseURL/part_8/tic_tac_toe_v7/board/board_cell_type.go
   // Возвращаем строковое представление фигуры
   func (f BoardField) String() string {
       switch f {
       case Cross:
           return "X"
       case Nought:
           return "O"
       default:
           return " "
       }
   }
