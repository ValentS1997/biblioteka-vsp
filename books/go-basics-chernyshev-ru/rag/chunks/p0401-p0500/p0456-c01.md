   func (r *SQLiteRepository) TaskDone(taskId int) error {
       if taskId = = 0 {
           return errors.New("invalid updated ID")
       }
       pjTask := &Task{ID: taskId}
       tx := r.db.Find(&pjTask)
       if tx.Error ! = nil {
           return tx.Error
       }
       pjTask.IsDone = true
       r.db.Save(&pjTask)

           return ErrUpdateFailed

   }
