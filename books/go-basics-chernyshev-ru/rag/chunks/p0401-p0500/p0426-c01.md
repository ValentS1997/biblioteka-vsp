8.4. Пример реализации TCP-чата
Создайте новый проект, но сохраните структуру предыдущих двух. В этом случае
мы реализуем сервер для TCP-чата и клиента, работающего с ним. Рассылка со
общений от клиентов будет не адресная, а широкополосная. То есть сообщение от
одного пользователя будет рассылаться всем.
  В файл client.go добавьте следующий код:

   // baseURL/part_8/8.4/tcp-chat/client/client.go
   package client

   import (
       "bufio"
       "fmt"
       "io"
       "log"
       "net"
       "os"
       "strings"
       "sync"
   )

   var waitingGr sync.WaitGroup

   func checkErr(err error) {
       if err ! = nil {
           log.Fatal(err)
       }
   }

   func sendMessage(connection net.Conn, waitingGr *sync.WaitGroup) {
       // горутина для работы с вводимыми сообщениями и их отправки
       defer waitingGr.Done()
       for {
           // считываем введенное сообщение
           reader := bufio.NewReader(os.Stdin)
           text, _ := reader.ReadString('\n')

           if strings.TrimSuffix(strings.TrimSpace(text), "\r\n") = = "/stop" {
               break // завершение работы клиента
           } else {
               _, err := connection.Write([]byte(text))
               checkErr(err)
           }
       }
   }
