   Осталось только написать реализацию запроса на изменение статуса задачи на
«Выполнено»:

   func doneTask(w http.ResponseWriter, r *http.Request) {
       w.Header().Set("Content-Type", "application/json; charset=UTF-8")
       queryStr := r.URL.Query()

       idStr, ok := queryStr["id"]
       if !ok {
           errorResponse(w, ErrorResponse{http.StatusBadRequest,
               "Validation Failed"})
           return
       }
       if len(idStr) = = 0 {
           errorResponse(w, ErrorResponse{http.StatusBadRequest,
               "Validation Failed"})
           return
       }

       id, err := strconv.Atoi(idStr[0])
       if err ! = nil {
           errorResponse(w, ErrorResponse{http.StatusBadRequest,
               "Validation Failed"})
           return
       }

       err = repository.TaskDone(id)
       if err ! = nil {
           errorResponse(w, ErrorResponse{http.StatusNotFound,
               "Tasks not found"})
           return
       }
       w.WriteHeader(http.StatusOK)
   }
   Веб-сервис практически готов к запуску! Не хватает только кода, связывающего
функционал работы с базой данных через ORM, и маршрутизатора запросов. Чтобы
исправить этот недочет, откройте файл main.go и внесите в него следующий код:

   // baseURL/part_8/8.5/golang/todo-service/main.go
   package main

   import (
       "golang/todo-service/db"
       "golang/todo-service/service"
       "log"
       "net/http"
   )

   const PORT = "8080"

   func main() {

       rep := db.NewSQLiteRepository()
       defer rep.Close()

       router := service.NewRouter(rep)
       log.Printf("Server started")
       log.Fatal(http.ListenAndServe(":"+PORT, router))
   }
