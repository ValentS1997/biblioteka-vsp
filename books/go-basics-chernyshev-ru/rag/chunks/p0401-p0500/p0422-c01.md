8.3. Пример клиента и сервера с установлением соединения
В этом случае воспользуемся предыдущей структурой у создаваемого нового про
екта. В файл client.go добавьте следующий код:

   // baseURL/part_8/8.3/tcp/client/client.go
   package client

   import (
       "bufio"
       "fmt"
       "log"
       "net"
   )

   func checkErr(err error) {
       if err ! = nil {
           log.Fatal(err)
       }
   }
