106       Глава 2   Функции, модули, пакеты и их тестирование

  Как видите, указатель можно использовать для того, чтобы работа в функции вы
полнялась с исходным значением переменной, а не его копией. Такое свойство указа
телей позволяет объявлять функции, которые явно не возвращают ни одного значения,
но на самом деле совершают эту операцию через указатель одного из аргументов:

   // baseURL/part_2/2.1/2.1.1/4.go
   package main

   import "fmt"

   func main() {
       a, b, rez := 10, 13, 0
       add(a, b, &rez)
       fmt.Printf("%d + %d = %d\n", a, b, rez)
   }

   func add(a, b int, rez *int) {
       *rez = a + b
   }

   // 10 + 13 = 23
  В следующем примере напишем две функции, которые проверяют вхождение
элемента в срез и разными способами возвращают значение в основную программу:

   // baseURL/part_2/2.1/2.1.1/5.go
   package main

   import "fmt"

   func find1(slice *[]int, value int, check *bool) {
       *check = false
       for i := range *slice {
           if (*slice)[i] = = value {
               *check = true
               break
           }
       }
   }

   func find2(slice *[]int, value int) bool {
       for i := range *slice {
           if (*slice)[i] = = value {
               return true
           }
       }
       return false
   }

   func main() {
       slice := []int{2, 4, 5, 7, 103, 55}
       var check bool
       value := 2
       find1(&slice, value, &check)
       fmt.Printf("%d contains in slice? %t\n", value, check)
       value = 22
       fmt.Printf("%d contains in slice? %t\n", value, find2(&slice, value))
   }

   // 2 contains in slice? true
   // 22 contains in slice? false
