   Структура может содержать только одно анонимное поле для каждого из типов
данных. Чтобы проверить это утверждение, удалите еще одно из имен полей и по
пытайтесь запустить приложение.

3.1.4. Структура как аргумент функции
При использовании структуры в качестве входного аргумента функции следует учи
тывать, что она будет передаваться по значению, то есть копироваться. Аналогичное
поведение наблюдается и в момент присваивания структуры другой переменной
этого типа данных. Чтобы в функции работать не с копией, а с ее отображением
в памяти приложения, передавайте ее через указатель:

   // baseURL/part_3/3.1/9.go
   package main

   import "fmt"

   type employee struct {
       name string
       age  uint8
   }

   func valueFunc(emp employee) {
       emp.name = "O_O"
       fmt.Printf("Copy value: %+v\n", emp)
   }

   func pointFunc(emp *employee) {
       emp.name = "^_^"
       fmt.Printf("Point to value: %+v\n", emp)
   }

   func main() {
       emp := employee{
           name: "Tom",
           age:  45,
       }
       newEmployee := emp
       newEmployee.age = 22
       fmt.Printf("newEmployee = %+v\n", newEmployee)
       fmt.Printf("emp = %+v\n", emp)

       valueFunc(emp) // передача в функцию по значению
       fmt.Printf("emp = %+v\n", emp)
       pointFunc(&emp) // передача в функцию по указателю
       fmt.Printf("emp = %+v\n", emp)
   }
