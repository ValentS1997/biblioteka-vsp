   func printShapeName(name shape.IShapeName) {

   func getIShapeFromRectangle(rectangle *shape.Rectangle) shape.IShape {

   func main() {
       rectangle := shape.NewRectangle(10, 6, shape.Point{X: 6, Y: -8}, "Gold")
       triangle := shape.NewTriangle(6, 10, 8, shape.Point{X: -85, Y: 15}, "Green")

       printShapeName(rectangle)                     // неявное приведение
       //printShapeName(shape.IShapeName(rectangle)) // явное приведение

       printShapeName(triangle)
       ishape := getIShapeFromRectangle(rectangle)
       fmt.Printf("Shape center: %v\n", ishape.GetCenter())
       fmt.Printf("Shape perimeter: %v\n", ishape.GetPerimeter())
   }

  Как видно из примера, при передаче структуры на вход функции, аргумент ко
торой имеет интерфейсный тип, реализуемый этой структурой, выполнять явное
приведение не требуется. То же относится и к возвращаемому функцией значению.
Компилятор все сделает сам за вас.
  Еще один способ использования интерфейса — итерация по структурам, по
мещенным в срез интерфейсного типа, то есть приведенным к интерфейсу и затем
помещенным в срез. Для чего это делается? Представьте, что работаете с большим
количеством структур различного типа, у которых необходимо вызвать один
и тот же метод. Заводить под каждый тип структуры отдельный срез и писать код
множества циклов с вызовом одного и того же метода — нерационально. Именно
в такой ситуации и можно использовать интерфейс:
