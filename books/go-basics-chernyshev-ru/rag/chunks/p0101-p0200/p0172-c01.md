3.7. Интерфейсы и полиморфизм времени исполнения      171

                    Рис. 3.2. Пример сокрытия методов после приведения к интерфейсу

  Кроме того, один интерфейс можно приводить к другому. Важное условие со
вершения такой операции — наличие у них методов с одинаковым описанием:

   // baseURL/part_3/3.7/golang/factory/main.go
   func main() {
       rectangle := shape.NewRectangle(10, 6, shape.Point{X: 6, Y: -8}, "Gold")

       // приведение Rectangle к интерфейсу
       ishape := shape.IShape(rectangle)

       // приведение интерфейса IShape к интерфейсу IShapeArea
       iShapeArea := shape.IShapeArea(ishape)
       fmt.Printf("IShapeArea: %+v\n", iShapeArea.GetArea())

       // приведение интерфейса IShape к интерфейсу IShapePerimeter
       iShapePerimeter := shape.IShapePerimeter(ishape)
       fmt.Printf("IShapePerimeter: %+v\n",
                   iShapePerimeter.GetPerimeter())
   }
   // IShapeArea: 60
   // IShapePerimeter: 32
   Обратное приведение более специализированного интерфейса к общему воз
можно при использовании следующего кода:

   // baseURL/part_3/3.7/golang/factory/main.go
   func main() {
       rectangle := shape.NewRectangle(10, 6, shape.Point{X: 6, Y: -8}, "Gold")

       // приведение Rectangle к интерфейсу
       ishape := shape.IShape(rectangle)

       // приведение интерфейса IShape к интерфейсу IShapeArea
       iShapeArea := shape.IShapeArea(ishape)
       fmt.Printf("IShapeArea: %+v\n", iShapeArea.GetArea())

       // явное обратное приведение интерфейса IShapePerimeter
       // к интерфейсу IShape невозможно

       // ishape = shape.IShape(iShapePerimeter) //error:

       // правильное приведение интерфейса IShapeArea к интерфейсу IShape
       ishape = iShapeArea.(*shape.Rectangle)
       // аналогично
       // newIshape := shape.IShape(iShapeArea.(*shape.Rectangle))
       fmt.Printf("IShape center: %+v\n", ishape.GetCenter())
