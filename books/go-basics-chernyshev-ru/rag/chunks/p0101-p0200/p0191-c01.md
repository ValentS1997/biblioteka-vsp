190       Глава 3   Объектно-ориентированное программирование в Go

  В файле board.go объявим структуру Board, возложив на нее обязанность
отвечать за все действия, связанные с игровым полем. По сути, мы переместим
в этот файл все функции, ранее работавшие с игровой доской, сделав их методами
структуры:

   // baseURL/part_3/tic_tac_toe_v2/game/board.go
   package game

   import (
       "fmt"
   )

   const (
       BoardDefaultSize int = 3
       BoardMinSize     int = 3
       BoardMaxSize     int = 9
   )

   type Board struct {
       board [][]BoardField
       size  int
   }

   func NewBoard(size int) *Board {
       board := make([][]BoardField, size)
       for i := range board {
           board[i] = make([]BoardField, size)
       }
       return &Board{board: board, size: size}
   }

   // Отображение игрового поля
   func (b *Board) printBoard() {
       fmt.Print("  ")
       for i := range b.size {
           fmt.Printf("%d ", i+1)
       }
       fmt.Println()
       for i := range b.size {
           fmt.Printf("%d ", i+1)
           for j := range b.size {
               switch b.board[i][j] {
               case empty:
                   fmt.Print(". ")
               case cross:
                   fmt.Print("X ")
               case nought:
                   fmt.Print("O ")
               }
           }
           fmt.Println()
       }
   }

   // Проверка возможности выполнения хода
   func (b *Board) makeMove(x, y int) bool {
       return b.board[x][y] = = empty
   }
