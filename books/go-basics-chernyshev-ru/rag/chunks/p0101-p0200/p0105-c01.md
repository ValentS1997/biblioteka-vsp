104       Глава 2   Функции, модули, пакеты и их тестирование

   Иногда функции могут использоваться для создания и инициализации какой-
либо громоздкой структуры данных, объекта или группы объектов. В таких случаях,
чтобы возвращаемое функцией значение не копировалось, вызывая замедление
работы приложения, принято использовать указатели. То есть функция должна
возвращать указатель на тип переменной:

   package main

   import (
       "fmt"
       "math/rand"
       "time"
   )

   // генератор случайных чисел
   var generator = rand.New(rand.NewSource(time.Now().UnixNano()))

   func sliceCreator1(size int16) (slice *[]int16) {
       slice = new([]int16)
       *slice = make([]int16, size, size+10)
       for i := range *slice {
           (*slice)[i] += int16(generator.Int())
       }
       return
   }

   func sliceCreator2(size int16) *[]int16 {
       slice := make([]int16, size, size+10)
       for i := range slice {
           slice[i] += int16(generator.Int())
       }
       return &slice
   }

   func main() {
       slice := sliceCreator1(7)
       fmt.Println(*slice)
       slice = sliceCreator2(5)
       fmt.Println(*slice)
   }
   // [–22201 –30245 6682 28346 23159 –28589 16762]
   // [3896 28732 –30834 1722 –14725]

2.1.1. Способы передачи аргументов в функцию
Существует два способа передачи аргументов в функцию: по значению и указатель
на переменную. В первом случае значение копируется и все последующие опера
ции в функции выполняются с копией, что никак не влияет на исходное значение,
хранящееся в переменной:

   // baseURL/part_2/2.1/2.1.1/1.go
   package main

   import "fmt"

   func main() {
       array := [4]float64{2.4, 5.6, 8.1, 9.22}
       myFunc(array)
       fmt.Printf("Array in main: %v\n", array)
   }
