2.1. Функции в Go      103

       fmt.Printf("a + b = %d, a > b = %t", rez, check)
   }

   func add(a, b int) (rez int, check bool) {
       rez = a + b
       check = a > b
       return // аналогично return rez, check

   }
   // a + b = 18, a > b = true
   Более заумное название таких функций — «функции, возвращающие имено
ванные значения».
  У функций в Go есть еще одна особенность: они могут содержать различ
ное количество входных аргументов (срез аргументов одного типа данных). Раз
берем такой вариант на примере суммирования значений, подаваемых на вход
функции:

   package main

   import "fmt"

   func main() {
       fmt.Printf("Sum = %d", myFunc(3, 5, 9, 25, –10))
   }

   func myFunc(values ...int) (sum int) {
       // переменную values рассматриваем как срез []int
       for _, value := range values {
           sum += value
       }
       return
   }
   // Sum = 32
  Если в функцию необходимо добавить аргументы другого типа данных, то их
нужно указать до values ...int и они не должны представлять собой объявление
из произвольного количества элементов:

   func myFunc(newValues ...float64, values ...int) (sum int) // 1 – error
   func myFunc(values ...int, str string) (sum int)  // 2 – error
   func myFunc(str string, values ...int) (sum int)  // 3 – ok
  Еще один способ использования функции — объявление и вызов на месте.
В этом случае функция будет считаться анонимной, так как нет необходимости
давать ей имя:

   package main

   import "fmt"

   func main() {
       sum := func(values ...int) (sum int) { // объявление функции
           // переменную values рассматриваем как срез []int
           for _, value := range values {
               sum += value
           }
           return
       }(3, 5, 9, 25, –10) // вызов функции
       fmt.Printf("Sum = %d", sum) // Sum = 32
   }
