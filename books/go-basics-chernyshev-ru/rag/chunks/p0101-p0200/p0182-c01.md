  Чтобы избежать падения приложения при таком неверном утверждении, вос
пользуемся следующим вариантом проверки:

   // baseURL/part_3/3.9/4.go
   func main() {
       var firstInterface interface{} = shape{name: "Cube"}
       value, ok := firstInterface.(string)

       if ok {
           fmt.Println(value) // работаем со значением
       } else {
           fmt.Println("Wrong type assertion!")
       }

       newValue, newOk := firstInterface.(shape)

       if newOk {
           fmt.Println(newValue.getName())
       } else {
           fmt.Println("Wrong type assertion!")
       }
   }
   // Wrong type assertion!
   // Cube
   Возможно, у вас возник вопрос: «Зачем разработчики наделили интерфейсы
таким сложным поведением?» Дело в том, что интерфейс, как абстрактный тип
данных, отвечает за бо́льшую часть реализации полиморфизма в Go.
  Еще одной особенностью интерфейса является то, что он может выступать
в качестве значения, подаваемого на вход условной конструкции switch. В случае
применения с интерфейсами ее принято называть type-switches, так как в блоке
case могут использоваться только типы:

   // baseURL/part_3/3.9/5.go
   package main

   import "fmt"

   type shape struct{}
   type car struct{}

   func checkType(i interface{}) {
       switch i.(type) {
       case shape:
           fmt.Println("It is shape")
       case car:
           fmt.Println("It is car")
       default:
           fmt.Println("What is this pokemon?")
       }
   }

   func main() {
       checkType(shape{})
       checkType(car{})
       checkType(6)
   }
