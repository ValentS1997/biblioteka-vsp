  Кроме того, замыкания можно использовать в программах, которые должны
динамически генерировать обработчики событий в ответ на условия, сложившиеся
во время выполнения.
  В качестве еще одного примера замыкания реализуем функцию, принимающую сте
пень, в которую будут возводиться числа, подаваемые на вход возвращаемой функции:

   // baseURL/part_2/2.1/2.1.7/2.go
   package main

   import (
       "fmt"
       "math"
   )

   type MyPow func(value int) int

   func degree(degree int) MyPow {
       return func(value int) int {
           return int(math.Pow(float64(value), float64(degree)))
       }
   }

   func main() {
       calculation := degree(3)
       fmt.Println(calculation(3)) // 27
       fmt.Println(calculation(4)) // 64
       calculation = degree(9)
       fmt.Println(calculation(4)) // 262144
       fmt.Println(calculation(7)) // 40353607
   }
   Теперь изменим пример так, чтобы в функции main объявлялась переменная
и функция, которая захватывает и изменяет ее значение, а также выступает входным
аргументом другой функции, которая, в свою очередь, путем замыкания вернет еще
одну функцию, с которой можно работать в дальнейшем:

   // baseURL/part_2/2.1/2.1.7/3.go
   package main

   import "fmt"

   type InputFunc func(int, int) int
   type OutputFunc func(int) int

   func myClosure(a, b int, foo InputFunc) OutputFunc {
       return func(value int) int {
           return value - foo(a, b)
       }
   }
