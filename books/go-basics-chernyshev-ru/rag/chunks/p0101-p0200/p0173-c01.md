       // правильное приведение интерфейса IShapeArea

   IShapeArea: 60

  Для следующего примера объявим в ishape.go еще один интерфейс:

   // baseURL/part_3/3.7/golang/factory/shape/ishape.go
   type IShapeName interface {
       GetName() string
   }
   После чего в коде исполняемого пакета выполним приведение к нему структуры
Rectangle и Triangle:

   // baseURL/part_3/3.7/golang/factory/main.go
   package main

   import (
       "fmt"
       "golang/factory/shape"
   )

   func main() {
       rectangle := shape.NewRectangle(10, 6, shape.Point{X: 6, Y: -8}, "Gold")
       triangle := shape.NewTriangle(6, 10, 8, shape.Point{X: -85, Y: 15}, "Green")

       // приведение Triangle к интерфейсу
       iShapeName := shape.IShapeName(triangle)
       fmt.Printf("IShapeName: %v\n", iShapeName.GetName())

       // приведение Rectangle к интерфейсу
       iShapeName = shape.IShapeName(rectangle)
       fmt.Printf("IShapeName: %v\n", iShapeName.GetName())
   }
   // IShapeName: Triangle
   // IShapeName: Super-puper Rectangle
   Обратите внимание на вывод имен в терминал. В первом случае через интер
фейс был вызван метод базовой структуры, а во втором — переопределенный метод
GetName структуры Rectangle. Такое поведение делает процесс написания кода при
ложений еще более гибким.
  Интерфейсы можно использовать в качестве входных аргументов функций
и методов или как возвращаемое ими значение:

   package main

   import (
       "fmt"
       "golang/factory/shape"
