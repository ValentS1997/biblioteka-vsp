110       Глава 2   Функции, модули, пакеты и их тестирование

   func add(a, b int) int {
       return a + b
   }

   func sub(c, a, b int, addFunc func(a int, b int) int) int {
       return c - addFunc(a, b)
   }

   func main() {
       myFunc := add                      // var myFunc func(a int, b int) int = add
       fmt.Println(sub(4, 3, 10, add))    // –9
       fmt.Println(sub(4, 3, 10, myFunc)) // –9
   }
  Чтобы записать объявление функции sub более компактно, используем ключе
вое слово type, позволяющее объявлять псевдонимы (alias declarations). Укажем,
что типу MyFunctionAdd соответствуют функции, которые принимают на вход два
значения типа int и возвращают значение аналогичного типа данных:

   // baseURL/part_2/2.1/2.1.5/2.go
   package main

   import "fmt"

   type MyFunctionAdd func(a int, b int) int

   func add(a, b int) int {
       return a + b
   }

   func sub(c, a, b int, addFunc MyFunctionAdd) int {
       return c – addFunc(a, b)
   }

   func main() {
       myFunc := add                      // var myFunc func(a int, b int) int = add
       fmt.Println(sub(4, 3, 10, add))    // –9
       fmt.Println(sub(4, 3, 10, myFunc)) // –9
   }

2.1.6. Анонимные функции
Эти функции не содержат явных названий и объявляются следующим образом:

   func([имяАргумента1 ТипВходногогоАргумента1, …, n]) [(возвращаемый тип данных)] {
       // тело функции
       [return возвращаемое значение]
   }
   Объявляемая анонимная функция должна присваиваться переменной, возвра
щаться другой функцией, подаваться на вход функции или вызываться на месте.
В качестве примера реализуем функцию add как анонимную, подав ее на вход
функции sub:
