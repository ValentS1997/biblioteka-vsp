   type shape struct{}
   type car struct{}

   func isEqual(i interface{}, j interface{}) {

       } else {
           fmt.Println("Inequal")
       }
   }

   func main() {
       isEqual(shape{}, shape{}) // Equal
       isEqual(shape{}, car{}) // Inequal

       var firstInterface interface{} // по умолчанию — nil
