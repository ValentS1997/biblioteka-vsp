               Рис. 1.14. Изменение структуры среза при добавлении новых элементов

Этот же оператор можно использовать и для удаления данных из среза:
package main

import "fmt"

func main() {
    slice := []string{"1", "2", "Hello", "4", "Hi", "6", "Alex", "8", "9"}

    fmt.Println(slice) // [1 2 Hello 4 Hi 6 Alex 8 9]
    // удаляем третий элемент в срезе
    slice = append(slice[:2], slice[2+1:]...)
    fmt.Println(slice) // [1 2 4 Hi 6 Alex 8 9]

    // удаляем элементы из середины среза
    slice = append(slice[:3], slice[6:]...)
    fmt.Println(slice) // [1 2 4 8 9]

    // удаляем первый элемент среза
    slice = slice[1:]
    fmt.Println(slice) // [2 4 8 9]

    // удаляем последний элемент среза
    slice = slice[:len(slice)–1]
    fmt.Println(slice) // [2 4 8]
}
