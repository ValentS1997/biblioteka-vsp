   Проверить строку на равенство, где непринципиален регистр символов, можно
с помощью функции EqualFold (s, t string):

   // baseURL/part_1/1.15/12.go
   package main

   import (
       "fmt"
       "strings"
   )

   func main() {
       str := "Hello mad world!"
       fmt.Println(strings.EqualFold(str, "hello mad world!")) // true
       fmt.Println(strings.EqualFold(str, "HELLO mad world!")) // true
       fmt.Println(strings.EqualFold(str, "Oo"))               // false
   }
  Для проверки того, заканчивается или начинается строка на определенные
символы (подстроки), используются функции HasPrefix и HasSuffix:

   // baseURL/part_1/1.15/13.go
   package main

   import (
       "fmt"
       "strings"
   )

   func main() {
       str := "Hello mad world!"
       fmt.Println(strings.HasPrefix(str, "He"))  // true
       fmt.Println(strings.HasPrefix(str, "he"))  // false
       fmt.Println(strings.HasPrefix(str, "mad")) // false

       fmt.Println(strings.HasSuffix(str, "!"))   // true
       fmt.Println(strings.HasSuffix(str, "mad")) // false
   }
