1.10. Условные конструкции в Go      53

   package main

   import "fmt"

   func main() {
       var value int = 10
       if value > 5 && value % 2 = = 0 {
           fmt.Println("Значение удовлетворяет условиям")
       } else {
           fmt.Println("Значение не удовлетворяет условиям")
       }
   }

   // Значение удовлетворяет условиям
  А если нужно выполнить действие в двух разных случаях, которые могут быть
равновероятны, то используйте логическое «ИЛИ»:

   package main

   import "fmt"

   func main() {
       var value int = -10
       if (value > 5 && value%2 = = 0) || value < 0 {
           fmt.Println("Значение удовлетворяет условиям")
       } else {
           fmt.Println("Значение не удовлетворяет условиям")
       }
   }

   // Значение удовлетворяет условиям
   Условные конструкции могут быть различной степени вложенности, когда после
проверки условия на истинность в теле блока if или else будет идти следующий
блок условий:

   // baseURL/part_1/1.10/1.10.1/4.go
   package main

   import "fmt"

   func main() {
       var value int = 15
       if value > 30 {
           if value < 50 {
               fmt.Println("30 < value < 50")
           } else {
               fmt.Println("30 < value > = 50")
           }
       } else {
           if value > 10 {
               fmt.Println("30 > = value > = 10")
           } else {
               fmt.Println("30 > = value < 10")
           }
       }
   }

   // 30 > = value > = 10
