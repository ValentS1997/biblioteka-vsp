   func main() {
       str1 := "hello"
       fmt.Println(str1) // hello

       byteArray := []byte(str1)
       fmt.Println(byteArray) // [104 101 108 108 111]

       str2 := string(byteArray)
       fmt.Println(str2) // hello
   }
  В очень редких случаях разработчику необходимо преобразовывать логический
тип данных в цифровой и наоборот. Это делается путем проверок и использования
условного оператора if:

   // baseURL/part_1/1.13/10.go
   package main

   import "fmt"

   func main() {
       myBool := true
       var value int
       if myBool {
           value = 1
       } else {
           value = 0
       }
       fmt.Println("Value = ", value) // Value =  1

       myBool2 := value = = 1
       fmt.Println("myBool2 value = ", myBool2) // myBool2 value = true
   }

1.14. Ввод данных, отладка и форматирование вывода
В некоторых случаях необходимо ввести с клавиатуры некий код прямо во время
работы консольного приложения. Ниже показан пример такого кода. К сожалению,
VS Code не даст корректно выполнить его в режиме отладки из-за отсутствия
дополнительных настроек. Поэтому для начала запустим программу с помощью
команды go run main.go через терминал:

   // baseURL/part_1/1.14/1.go
   package main

   import (
       "fmt"
       "os"
   )

   func main() {
       var cityName string
       var age int
       fmt.Print("Введите название города, в котором проживаете: ")
       fmt.Fscan(os.Stdin, &cityName)

       fmt.Print("Введите возраст города: ")
       fmt.Fscan(os.Stdin, &age)

       fmt.Println(cityName, age)
   }
