78       Глава 1   Синтаксис, встроенные типы данных и управляющие конструкции Go

  Для начала представим, что пользователь, написавший код, любил в начале или
в конце предложения вставлять пробелы. Очистить такие строки от обрамляющих
пробелов можно с помощью функций семейства Trim:

   // baseURL/part_1/1.15/1.go
   package main

   import (
       "fmt"
       "strings"
   )

   func main() {
       str := " !my string! "

       // удаляем пробелы с обеих сторон
       fmt.Println(strings.TrimSpace(str)) // !my string!

       // удаляем заданные символы в начале и в конце строки
       fmt.Println(strings.Trim(str, " ")) // !my string!
       fmt.Println(strings.Trim(str, " !")) // my string

       // удаляем заданные символы в начале строки
       fmt.Println(strings.TrimLeft(str, " ")) // !my string!
       fmt.Println(strings.TrimLeft(str, " m!")) // y string!

       // удаляем заданные символы в конце строки
       fmt.Println(strings.TrimRight(str, " ")) // !my string!
       fmt.Println(strings.TrimRight(str, "n!g ")) //  !my stri

       // удаляем заданный символ или подстроку в начале строки
       fmt.Println(strings.TrimPrefix(str, " !my")) // string!

       // удаляем заданный символ или подстроку в конце строки
       fmt.Println(strings.TrimSuffix(str, "g! ")) // !my strin
   }
  Если же программа не должна зависеть от того, в каком регистре пользователь
ввел данные, — приведите строку к верхнему или нижнему регистру:

   // baseURL/part_1/1.15/2.go
   package main

   import (
       "fmt"
       "strings"
   )

   func main() {
       str1 := "Hi, my dear friend!!!"
       str2 := strings.ToUpper(str1)
       fmt.Println(str2) // HI, MY DEAR FRIEND!!!
       fmt.Println(str1) // Hi, my dear friend!!!

       fmt.Println(strings.ToLower(str2)) // hi, my dear friend!!!
       fmt.Println(strings.ToTitle(str1)) // HI, MY DEAR FRIEND!!!
   }
