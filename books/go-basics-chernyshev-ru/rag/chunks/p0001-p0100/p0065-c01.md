64       Глава 1   Синтаксис, встроенные типы данных и управляющие конструкции Go

   Аналогичное поведение будет наблюдаться и при приведении вещественного
значения к целочисленному:

   // baseURL/part_1/1.13/3.go
   package main

   import "fmt"

   func main() {
       var val1 int = 5
       var val2 float32 = 5.32
       fmt.Printf("%.1f\n", float32(val1)) // 5.0
       // Printf рассматривается в следующем разделе

       fmt.Println(int(val2))              // 5
   }
  Кроме того, следует обратить внимание на переполнение, когда результат
сложения двух значений может выйти за разрядную сетку их базового типа
данных:

   // baseURL/part_1/1.13/4.go
   package main

   import "fmt"

   func main() {
       var val1 uint8 = 5 // от 0 до 255
       var val2 uint8 = 255
       fmt.Println(val2 + val1) // 4
       rez1 := uint16(val1) + uint16(val2)
       fmt.Println(rez1) // 260
   }
  При приведении беззнакового целочисленного значения к знаковому могут
появиться отрицательные значения, так как старший бит будет отвечать за знак
(0 — положительное число, 1 — отрицательное):

   // baseURL/part_1/1.13/5.go
   package main

   import "fmt"

   func main() {
       var val1 uint8 = 255 // 1111 1111 в двоичной системе счисления
       val2 := int8(val1)
       fmt.Println(val2) //–1, что равно 1111 1111 в двоичной системе счисления
   }

  При делении целых чисел результат также будет целочисленным:

   // baseURL/part_1/1.13/6.go
   package main

   import "fmt"

   func main() {
       val1 := 4 / 3
       fmt.Println(val1) // 1

       val2 := 4.0 / 3
       fmt.Println(val2) // 1.3333333333333333
   }
