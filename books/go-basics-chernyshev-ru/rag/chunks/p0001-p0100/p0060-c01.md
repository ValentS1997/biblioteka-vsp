   Такое же поведение характерно для continue при наличии вложенных циклов,
в то время как break передаст управление на строчку кода, следующую за телом
главного цикла:

   // baseURL/part_1/1.11/5.go
   package main

   import "fmt"

   func main() {
       for i := 0; i < 2; i+ + {
           for j := 0; j < 3; j+ + {
               if i = = j {
                   continue
               }
               fmt.Printf("%d %d || ", i, j)
           }
       }
       // 0 1 || 0 2 || 1 0 || 1 2 ||

       fmt.Println()
       for i := 0; i < 2; i+ + {
           for j := 0; j < 3; j+ + {
               if i = = j {
                   break
               }
               fmt.Printf("%d %d || ", i, j)
           }
       }
       fmt.Println("Oo")
       // 1 0 || Oo
   }
  Еще одна область применения цикла for — итерирование по коллекциям:
массивам, срезам, строкам и таблицам (map). Под итерированием понимается по
следовательный поэлементный обход структуры данных:

   // baseURL/part_1/1.11/6.go
   package main

   import "fmt"

   func main() {
       array := [4]int{2, 5, 6, 0}
       for i := 0; i < len(array); i+ + {
           array[i] += 3
           fmt.Printf("%d ", array[i])
       }
   }
   // 5 8 9 3
