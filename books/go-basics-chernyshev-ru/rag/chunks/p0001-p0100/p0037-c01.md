   Добавлять новые элементы можно с помощью функции append. Когда при ее
использовании длина среза становится больше емкости, система создаст новый
срез двойной емкости и в него скопируются элементы текущего:

   package main

   import "fmt"

   func main() {
       slice := make([]int, 2)
       fmt.Println(slice) // [0 0]
       fmt.Println(len(slice)) // длина = 2
       fmt.Println(cap(slice)) // емкость = 2

       // добавляем элемент со значением 3 в конец среза
       slice = append(slice, 3)

       fmt.Println(slice) // [0 0 3]
       fmt.Println(len(slice)) // длина = 3
       fmt.Println(cap(slice)) // емкость = 4

       slice = append(slice, 4, 5, 6, 7, 8)

       fmt.Println(slice)      // [0 0 3 4 5 6 7 8]
       fmt.Println(len(slice)) // длина = 8
       fmt.Println(cap(slice)) // емкость = 8
   }
  На рис. 1.14 показано, как менялись внутренние значения среза.
  Срезы можно создать из массива или из другого среза. Для этого использу
ется оператор [n:l], где n — индекс элемента, который будет первым элементом
в создаваемом срезе, а l — индекс, до которого будет формироваться новый срез.
Например, mySlice[3:10] означает, что из среза mySlice будет сформирован новый
срез, состоящий из элементов предыдущего, расположенных с третьего по девятый
индекс включительно. Как n, так и l можно опускать:

   package main

   import "fmt"

   func main() {
       slice := []int{1, 2, 3, 4, 5, 6, 7, 8, 9}
       array := [9]int{1, 2, 3, 4, 5, 6, 7, 8, 9}

       // создание среза с индекса 3 по конец массива
       slice1 := slice[3:]
       fmt.Println(slice1) // [4 5 6 7 8 9]

       // создание среза из элементов с третьего по шестой индекс
       slice2 := array[3:7]
       fmt.Println(slice2) // [4 5 6 7]

       // создание среза из элементов с нулевого по четвертый индекс
       slice3 := array[:5]
       fmt.Println(slice3) // [1 2 3 4 5]
