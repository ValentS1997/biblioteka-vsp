  Как и в случае с оператором if, при использовании конструкции switch–case
вы можете объявить и инициализировать переменную. Но будьте внимательными
и давайте ей имя, которое отличается от уже существующих, поскольку она будет
локальной для конструкции и заменит собой объявленную ранее:

   // baseURL/part_1/1.10/1.10.2/2.go
   package main

   import "fmt"

   func main() {
       value := 2
       switch value := 4; value {
       case 2:
           fmt.Println("2")
           fmt.Printf("%d + 2 = %d", value, value+2)
       case 4:
           fmt.Println("4")
       default:
           fmt.Println("8")
       }
       fmt.Printf("value = %d", value)
   }

   // 4
   // value = 2
   Если по нескольким значениям необходимо выполнить один и тот же блок кода,
то перечислите их через запятую после case:

   // baseURL/part_1/1.10/1.10.2/3.go
   package main

   import "fmt"

   func main() {
       name := "Alex"
       switch name {
       case "Stanislav":
           fmt.Println("Admin")
       case "Maxim", "Alex", "Bill":
           fmt.Println("Employee")
       }
   }

   // Employee
