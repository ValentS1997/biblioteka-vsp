56       Глава 1   Синтаксис, встроенные типы данных и управляющие конструкции Go

   import "fmt"

   func main() {
       name := "Stanislav"
       switch name {
       case "Stanislav":
           fmt.Println("Admin")
           fallthrough
       case "Maxim", "Alex", "Bill":
           fmt.Println("Employee")
       }
   }

   // Admin
   // Employee
  А когда нужно прекратить выполнение блока и передать управление на строку
кода, следующую за конструкцией switch–case, используйте ключевое слово break
(явно или после проверки условия в блоке if):

   // baseURL/part_1/1.10/1.10.2/5.go
   package main

   import "fmt"

   func main() {
       name := "Stanislav"
       switch name {
       case "Stanislav":
           fmt.Println("Admin")
           break
           fallthrough
       case "Maxim", "Alex", "Bill":
           fmt.Println("Employee")
       }
   }

   // Admin
  В отличие от некоторых других языков программирования, в Go switch–case
представляет собой более гибкий инструмент, что дает возможность использовать
условия при поиске совпадений:

   import "fmt"

   func main() {
       var value int = 5
       switch {
       case value > 30:
           switch {
           case value < 50:
               fmt.Println("30 < value < 50")
           default:
               fmt.Println("30 < value > = 50")
           }
       case value < 30:
           switch {
           case value > 10:
