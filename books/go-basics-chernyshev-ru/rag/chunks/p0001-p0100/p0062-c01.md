1.11. Цикл for      61

   func main() {
       s := "Привет"

       for i, r := range s {
           // i — номер индекса, где располагается первый байт символа
           // v — копия символа, приведенная к типу Rune
           fmt.Printf("Rune index: %d, Rune: %c (Hex: %x)\n", i, r, r)
       }
   } /*
   Rune index: 0, Rune: П (Hex: 41f)
   Rune index: 2, Rune: р (Hex: 440)
   Rune index: 4, Rune: и (Hex: 438)
   Rune index: 6, Rune: в (Hex: 432)
   Rune index: 8, Rune: е (Hex: 435)
   Rune index: 10, Rune: т (Hex: 442) */
  Исключением же может быть ситуация, если в строке только английские сим
волы:

   // baseURL/part_1/1.11/10.go
   package main

   import "fmt"

   func main() {
       s := "Hello"

       for i := 0; i < len(s); i+ + {
           fmt.Printf("Byte index: %d, Symbol: %c\n", i, s[i])
       }
   } /*
   Byte index: 0, Symbol: H
   Byte index: 1, Symbol: e
   Byte index: 2, Symbol: l
   Byte index: 3, Symbol: l
   Byte index: 4, Symbol: o */
  Помимо прочего, использование range позволяет писать куда меньше кода, когда
четко определено, какое количество раз должен выполниться цикл:

   // baseURL/part_1/1.11/11.go
   package main

   import "fmt"

   func main() {

       for i := 0; i < 5; i+ + {
           fmt.Printf("%d ", i) // 0 1 2 3 4
       }
       fmt.Println()

       // или
       for i := range 5 {
           fmt.Printf("%d ", i) // 0 1 2 3 4
       }
       fmt.Println()

       // или
       i := 0
       for range 5 {
           fmt.Printf("%d ", i) // 0 1 2 3 4
           i+ +
       }
   }
