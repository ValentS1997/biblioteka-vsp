1.13. Преобразование типов данных      65

  Для конвертации целых чисел в строки потребуется подключить пакет strconv
и использовать его функцию верхнего уровня — Itoa, а обратное преобразование
(из строки в число) можно выполнить с помощью функции Atoi:

   // baseURL/part_1/1.13/7.go
   package main

   import (
       "fmt"
       "reflect"
       "strconv"
   )

   func main() {
       val1 := 33
       var str string = strconv.Itoa(val1)
       fmt.Println(str) // 33
       val2, err := strconv.Atoi(str)
       if err = = nil { // если получилось преобразовать, то err = = nil
           fmt.Println(val2) // 33
           fmt.Println(reflect.TypeOf(val2)) // int
       }
       val2, err = strconv.Atoi("ee")
       if err ! = nil {
           fmt.Println(val2) // 0
       }

       val3 := 100
       fmt.Println("Rezult: " + strconv.Itoa(val3)) // Rezult: 100
   }
   Конвертировать значение вещественного типа данных можно с помощью функ
ции верхнего уровня из пакета strconv — FormatFloat (либо функции Sprintf из
пакета fmt), а для обратной операции необходимо использовать ParseFloat:

   // baseURL/part_1/1.13/8.go
   package main

   import (
       "fmt"
       "reflect"
       "strconv"
   )

   func main() {
       val1 := 33.321
       str1 := strconv.FormatFloat(val1, 'f', –1, 64)
       // равносильно
       str2 := fmt.Sprintf("%.3f", val1)
       fmt.Println(reflect.TypeOf(str2) , str2) // string 33.321
       fmt.Println(reflect.TypeOf(str1), str1) // string 33.321
       val2, _ := strconv.ParseFloat(str2, 64)
       fmt.Println(reflect.TypeOf(val2), val2) // float64 33.321
   }
   Более подробно входные аргументы функции FormatFloat и их назначение опи
саны в официальной документации пакета strconv.
