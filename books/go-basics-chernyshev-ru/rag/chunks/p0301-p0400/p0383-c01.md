   // Структура Person представляет владельца банковского счета
   type Person struct {
       ID   uint64
       Name string
       Age  uint8
   }

   // Структура Account представляет банковский счет с потокобезопасными операциями
   type Account struct {
       id      uint64
       owner   Person
       balance int
       // Мьютекс для защиты баланса при параллельном доступе
       lock    sync.RWMutex
   }

   // Метод для снятия денег со счета с обеспечением потокобезопасности
   func (a *Account) WithdrawMoney(amount int) {
       a.lock.Lock()
       defer a.lock.Unlock()
       a.balance -= amount
   }

   // Метод для зачисления денег на счет с обеспечением потокобезопасности
   func (a *Account) DepositMoney(amount int) {
       a.lock.Lock()
       defer a.lock.Unlock()
       a.balance += amount
   }

   // Метод для получения текущего баланса с использованием блокировки для чтения
   func (a *Account) GetBalance() int {
       a.lock.RLock()
       defer a.lock.RUnlock()
       return a.balance
   }
