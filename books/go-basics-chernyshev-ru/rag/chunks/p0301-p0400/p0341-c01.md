6.3.3. Ручное управление транзакцией
Модуль gorm поддерживает и прямой вызов функций управления транзакциями
(фиксация изменений/откат). За раз можно либо откатить изменения, либо при
нять их. При этом до момента фиксации изменений они не будут отображаться
в таблице базы:

   func main() {
       firstProject := Project{
           ID: 3,
       }

       db, err := connectionToBD(dbName)
       if err ! = nil {
           log.Fatal(err)
       }

       db.Find(&firstProject)
       fmt.Println("*********Projects*********")
       fmt.Printf("%+v\n", firstProject)
       tx := db.Begin()
       tx.Create(&ProjectTask{
           Name:        "Брусья",
           Description: "3 подхода по 15",
           Priority:    4,
           Project:     &firstProject,
       })
       tx.Create(&ProjectTask{
           Name:        "Турник",
           Description: "4 подхода по 1",
           Priority:    1,
           Project:     &firstProject,
       })
       tx.Commit() // фиксация изменений

       tx = db.Begin()

       tx.Create(&ProjectTask{
           Name:        "Турник",
           Description: "4 подхода по 1",
           Priority:    1,
           Project:     &firstProject,
       })

       printAllProjectTasks(db, firstProject.ID)
       fmt.Println("------RollBack------")
       tx.Rollback()

       fmt.Println()
       fmt.Println("Финальное состояние таблицы Taks проекта 'Спорт' ")
       printAllProjectTasks(db, firstProject.ID)
   }
