       fmt.Print("Input task ID: ")

       id, err := getIntValueFromStd(bufio.NewReader(os.Stdin))
       if err ! = nil {
           printNotValidData()
           return
       }

       err = rep.TaskDone(id)
       if err ! = nil {
           fmt.Println(err)
           return
       }
       fmt.Println("Congratulations! Task done!")
   }
