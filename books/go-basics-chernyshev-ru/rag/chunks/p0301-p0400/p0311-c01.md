310       Глава 6   Работа с СУБД SQLite

6.1.3. Создание базы данных
База может находиться в директории с проектом либо нет. В первом случае с ней
сразу открывается соединение, а во втором — на разработчика ложится обязан
ность создания не только файла с базой, но и ее таблиц. Сделать это поможет
структура SQLiteRepository, которую мы объявим в файле db.go. При создании
ее нового экземпляра будет выполняться проверка на наличие в системе базы
с необходимым именем. Сама же структура SQLiteRepository — обертка над ти
пом *sql.DB. Она позволит вывести код на более высокий уровень абстракции,
оградив разработчиков, использующих эту структуру, от деталей реализации
работы с базой:

   // baseURL/part_6/6.1/golang/todo/db/db.go
   package db

   import (
       "database/sql"
       "fmt"
       "log"
       "os"
   )

   type SQLiteRepository struct {
       db *sql.DB
   }

   func createDB(pathToDB string) *sql.DB {
       db, err := sql.Open("sqlite3", pathToDB)
       if err ! = nil {
           log.Fatal(err)
       }
       db.Exec(ProjectTabelDefinition) // создание таблицы projects
       db.Exec(TaskTabelDefinition) // создание таблицы tasks
       return db
   }

   func NewSQLiteRepository() *SQLiteRepository {
       var db *sql.DB

       if _, err := os.Stat(dbName); os.IsNotExist(err) {
           db = createDB(dbName)
           fmt.Println("DB isn't exist")
       } else {
           db, err = sql.Open("sqlite3", dbName)
           if err ! = nil {
               log.Fatal(err)
           }
           fmt.Println("DB already exists")
       }

       return &SQLiteRepository{
           db: db,
       }
   }

   func (s *SQLiteRepository) Close() {
       s.db.Close()
   }
