7.5. Каналы      377

7.5.7. Использование канала с оператором цикла for…range
Цикл for…range позволяет считывать данные из канала, присваивая значения его
элементов переменной цикла. В качестве примера изменим код, в котором при
передаче данных между основной программной и горутиной использовался канал
с пользовательским типом данных:

   // baseURL/part_7/7.5/15.go
   package main

   import (
       "fmt"
   )

   type Rectangle struct {
       Width  uint
       Length uint
   }

   func (r *Rectangle) GetPerimeter() uint {
       return (r.Length + r.Width) * 2
   }

   func (r *Rectangle) GetArea() uint {
       return r.Length * r.Width
   }

   func calculate(in < -chan Rectangle, out chan< - float64) {
       var sum float64
       for i := range in {
           sum += float64(i.GetPerimeter())
       }
       out < - sum / float64(cap(in))
   }

   func main() {
       inChan := make(chan Rectangle, 3)
       outChan := make(chan float64)
       go calculate(inChan, outChan)

       inChan < - Rectangle{14, 6}
       inChan < - Rectangle{5, 21}
       close(inChan) // закрытие канала

       result := < -outChan
       fmt.Printf("Result = %v", result)
   }
   // Result = 30.666666666666668
   Использование такого цикла при чтении из открытого канала может привести
к взаимной блокировке, поэтому в главной функции main мы выполнили его за
крытие. Попробуйте закомментировать строку close(inChan), после чего снова
запустите приложение. Пока канал открыт, цикл for…range никогда не завершится.
Поэтому программа аварийно завершится с ошибкой: fatal error: all goroutines
are asleep - deadlock!
