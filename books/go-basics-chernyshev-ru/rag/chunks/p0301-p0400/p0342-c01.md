использовать в сочетании с отложенными функциями и способами обработки паник:

   func main() {
       db, err := connectionToBD(dbName)
       if err ! = nil {
           log.Fatal(err)
       }
       fmt.Println("Начальное состояние таблицы Projects")
       printAllProjects(db)

       CreateProjects(db)

       fmt.Println()
       fmt.Println("Финальное состояние таблицы Projects")

   }

   func CreateProjects(db *gorm.DB) error {
       tx := db.Begin()
       defer func() {
           if r := recover(); r ! = nil {
               tx.Rollback()
           }
       }()

       if err := tx.Error; err ! = nil {
           return err
       }

       if err := tx.Create(&Project{Name: "Oo"}).Error; err ! = nil {
           tx.Rollback()
           return err
       }

       if err := tx.Create(&Project{Name: "^_^"}).Error; err ! = nil {
           tx.Rollback()
           return err
       }

       return tx.Commit().Error
   }
