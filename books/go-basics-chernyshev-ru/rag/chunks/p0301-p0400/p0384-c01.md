// Функция для симуляции многократного зачисления денег на счет
// Выполняется в отдельной горутине и сообщает о завершении через WaitGroup
func DepositMoney(waitGroup *sync.WaitGroup, account *Account) {
    for i := 0; i < count; i+ + {
        account.DepositMoney(i)
        time.Sleep(time.Microsecond)
    }
    waitGroup.Done()
}

// Функция для непрерывного мониторинга и вывода баланса счета
// Работает в бесконечном цикле в отдельной горутине
func PrintBalance(number int, account *Account) {
    for {
        fmt.Printf("%d)Current account balance = %d\n",
            number, account.GetBalance())
        time.Sleep(150 * time.Millisecond)
    }
}

func main() {
    // Создаем группу ожидания для координации горутин
    var waitGroup sync.WaitGroup

    // Создаем новый банковский счет с начальным балансом 30000
    account := NewAccount(
        324234234,
        Person{
            ID:   233424,
            Name: "Alex",
            Age:  21,
        },
        30000,
    )
    // Регистрируем две операции в группе ожидания
    waitGroup.Add(2)
    // Запускаем операции снятия денег в отдельной горутине
    go WithdrawMoney(&waitGroup, account)
    // Запускаем операции зачисления денег в отдельной горутине
    go DepositMoney(&waitGroup, account)
    // Запускаем несколько мониторов баланса в отдельных горутинах
    for i := 0; i < = 3; i+ + {
        go PrintBalance(i, account)
        time.Sleep(50 * time.Millisecond)
    }
    // Ожидаем завершения всех операций снятия и зачисления
    waitGroup.Wait()
    // Выводим конечный баланс счета после всех операций
    fmt.Printf("Final   account balance = %d", account.GetBalance())
}
