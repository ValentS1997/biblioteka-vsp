380       Глава 7   Конкурентность в Go

  По сути, мьютекс можно представить в виде канала с одним элементом, и очень
часто в интернете встречаются старые примеры, в которых для синхронизации
записи между горутинами предлагается использовать именно каналы. Такое при
менение некорректно, но знать об этом полезно для более глубокого понимания
работы каналов. Поэтому в первом примере вместо мьютекса используем буфери
зованный канал с одним элементом:

   // baseURL/part_7/7.7/1.go
   package main

   import (
       "fmt"
       "sync"
   )

   var value int

   func increment(waitGroup *sync.WaitGroup, c chan bool) {
       c < - true // блокируем критический раздел
       value = value + 1
       < -c // разблокировка критического раздела
       waitGroup.Done()
   }

   func main() {
       var waitGroup sync.WaitGroup
       myChan := make(chan bool, 1)
       for i := 0; i < 1000; i+ + {
           waitGroup.Add(1)
           go increment(&waitGroup, myChan)
       }
       waitGroup.Wait()
       fmt.Println("Value after 1k increment  = ", value)
   }
   // Value after 1k increment  =  1000
