5.1. Операции с файлами      219

  Еще один способ построчно считать данные из файла — использовать функцию
NewScanner из пакета bufio:

   // baseURL/part_5/5.1/11.go
   package main

   import (
       "bufio"
       "fmt"
       "log"
       "os"
   )

   func main() {
       file, err := os.Open("pirates.txt")
       if err ! = nil {
           log.Fatal(err)
       }
       defer file.Close()

       scanner := bufio.NewScanner(file)
       for scanner.Scan() {
           fmt.Println(scanner.Text())
       }

       if err := scanner.Err(); err ! = nil {
           log.Fatal(err)
       }
   }
   // Пятнадцать человек на сундук мертвеца,
   // Йо-хо-хо, и бутылка рома,
  А используя свойства переменной scanner типа *bufio.Scanner, можно реали
зовать последовательное считывание каждого слова:

   // baseURL/part_5/5.1/12.go
   func main() {
       file, err := os.Open("pirates.txt")
       if err ! = nil {
           log.Fatal(err)
       }
       defer file.Close()

       scanner := bufio.NewScanner(file)
       scanner.Split(bufio.ScanWords)
       for scanner.Scan() {
           fmt.Println(scanner.Text())
       }

       if err := scanner.Err(); err ! = nil {
           log.Fatal(err)
       }
   }
   /*
   Пятнадцать
   человек
   на
   сундук
   мертвеца,
   Йо-хо-хо,
   и
   бутылка
   рома, */
