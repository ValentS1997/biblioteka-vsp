  Поле head будет хранить ссылку на вершину односвязного списка, а tail — на
его последний элемент. Добавлять всегда будем в конец списка, обновляя объект,
на который ссылается данное поле.
   Теперь объявим метод для добавления записи в таблицу и проверки на то, суще
ствует ли в ней запись с таким идентификатором. Обратите внимание, что в методе
Contains мы обходим список, начиная с его вершины и двигаясь до тех пор, пока
следующий за узлом элемент не будет хранить nil:

   func (t *Table[T]) Contains(data T) bool {
       user, ok := any(data).(*User)
       if !ok {
           return false
       }

       temp := t.head
       for temp ! = nil {
           if temp.Data.Check(FieldID, strconv.Itoa(user.ID())) {
               return true
           }
           temp = temp.Next
       }
       return false
   }

   func (t *Table[T]) Insert(data T) bool {
       if t.Contains(data) {
           return false
       }
       node := &Node[T]{Data: data}
       if t.head = = nil {
           t.head = node
           t.tail = node
       } else {
           t.tail.Next = node
           t.tail = node
       }
       return true
   }
