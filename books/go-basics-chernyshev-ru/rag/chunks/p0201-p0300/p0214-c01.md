5.1.2. Получение информации о файле

Для получения информации о файле используется функция Stat, возвращающая
значение типа os.FileInfo:

   // baseURL/part_5/5.1/2.go
   package main

   import (
       "fmt"
       "log"
       "os"
   )

   func main() {
       fileInfo, err := os.Stat("test.txt")
       if err ! = nil {
           log.Fatal(err)
       }
       fmt.Println("File name:", fileInfo.Name())
       fmt.Println("Size in bytes:", fileInfo.Size())
       fmt.Println("Permissions:", fileInfo.Mode())
       fmt.Println("Last modified:", fileInfo.ModTime())
       fmt.Println("Is Directory: ", fileInfo.IsDir())
   }
   /*
   File name: test.txt
   Size in bytes: 0
   Permissions: -rw-rw-rw-
   Last modified: 2022-06-15 17:38:20.8403944 +0300 MSK
   Is Directory:  false
   */
