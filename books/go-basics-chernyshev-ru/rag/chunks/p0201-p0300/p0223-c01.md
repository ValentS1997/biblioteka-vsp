222       Глава 5   Работа с файлами и каталогами. Сборка приложения

  В следующем примере создадим файл с использованием рассматриваемой
функции:

   // baseURL/part_5/5.1/13.go
   package main

   import (
       "log"
       "os"
   )

   func main() {
       firstText := "Пятнадцать человек на сундук мертвеца,\n"
       secondText := "Йо-хо-хо, и бутылка рома,\n"
       file, err := os.OpenFile("pirates2.txt", os.O_WRONLY|os.O_CREATE, 0666)
       if err ! = nil {
           log.Fatal(err)
       }
       defer file.Close()

       file.WriteString(firstText)
       file.WriteString(secondText)
   }
  Попробуем снова открыть этот файл и дописать в него недостающие строки
припева:

   // baseURL/part_5/5.1/14.go
   package main

   import (
       "log"
       "os"
   )

   func main() {
       firstText := "Пей, и дьявол тебя доведет до конца,\n"
       secondText := "Йо-хо-хо, и бутылка рома!"
       file, err := os.OpenFile("pirates2.txt", os.O_WRONLY, 0666)
       if err ! = nil {
           log.Fatal(err)
       }
       defer file.Close()

       file.WriteString(firstText)
       file.WriteString(secondText)
   }
   Результат будет неожиданным (рис. 5.3).
