       "fmt"

   type Actor struct {
       Name        string
       Age         int
       FilmsAmount int
       AboutActor  string
   }

   func main() {
       actor := Actor{
           Name:        "Tom Hanks",
           Age:         69,
           FilmsAmount: 70,
           AboutActor:  "Tom Hanks is an actor...",
       }

       actorJson, err := json.Marshal(actor) // сериализация
       if err ! = nil {
           log.Fatal(err)
       }

       fmt.Println(string(actorJson))
   }
   /* {"Name":"Tom Hanks","Age":69,"FilmsAmount":70,"AboutActor":"Tom Hanks is an
   actor..."}*/
  Такой вывод неудобен для чтения, если в структуре большое количество
полей. Чтобы улучшить его, замените функцию json.Marshal(actor) на json.Mar
shalIndent(actor, "", " "). После перезапуска приложения в терминал будет вы
водиться более удобочитаемый вариант:

   {
       "Name": "Tom Hanks",
       "Age": 69,
       "FilmsAmount": 70,
       "AboutActor": "Tom Hanks is an actor..."
   }
  Прежде чем переходить к рассмотрению десериализации, сохраним преобразо
ванную в JSON-формат структуру в файл actor.json:

   // baseURL/part_5/5.3/2.go
   func main() {
       actor := Actor{ /* без изменений */  }

       actorJson, err := json.MarshalIndent(actor, "", "  ")
       if err ! = nil {
           log.Fatal(err)
       }

       myFile, err := os.Create("actor.json")
       if err ! = nil {
           log.Fatal(err)
       }

       myFile.Write(actorJson)
       defer myFile.Close()
   }
