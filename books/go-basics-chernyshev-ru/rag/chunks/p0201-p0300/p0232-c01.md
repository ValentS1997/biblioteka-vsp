   func main() {
       os.Mkdir(tempDirName, 0777)

       // случайное имя временного каталога
       path := printPath(os.MkdirTemp(tempDirName, "*"))
       createdFile(path)

       // имя с использованием шаблона
       path = printPath(os.MkdirTemp(tempDirName, "temp*dir"))
       createdFile(path)

       // имя с использованием шаблона
       path = printPath(os.MkdirTemp(tempDirName, "tempdir"))
       createdFile(path)
       // создание временного каталога со случайным именем
       // в системной директории для временных файлов
       path = printPath(os.MkdirTemp("", "*"))
       createdFile(path)
   }
