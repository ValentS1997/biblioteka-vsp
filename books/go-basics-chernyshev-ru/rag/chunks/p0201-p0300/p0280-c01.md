   func mergeMaps(a, b map[string]interface{}) map[string]interface{} {

   Поскольку запуск или сборка проекта производится из корневой директории, то
в ней и будет размещаться файл .env, но это не запрещает вам хранить его в другом
месте. Достаточно будет немного изменить код, добавив путь к месторасположению
вашего конфигурационного файла. Что же касается наполнения самого файла, то
сделаем его таким:

   // baseURL/part_5/5.6/go_dotenv/.env
   ADMIN_ID=123456789
   ADMIN_PASSWORD=wtfPassword1234
   ADMIN_EMAIL=admin@localhost
   ADMIN_MODE=true
   Теперь можно считать данные из конфигурационного файла и вывести их в тер
минал. Откройте main.go и добавьте в него обращение к локальным переменным
окружения через экземпляр класса DotEnv — env:

   // baseURL/part_5/5.6/go_dotenv/main.go
   package main

   import (
       "fmt"
       "go_dotenv/dotenv"
       "log"
   )

   func main() {
       env := dotenv.NewDotEnv(false)
       err := env.Load(".env")
       if err ! = nil {
           log.Fatal(err)
       }

       adminID, err := env.GetInt("ADMIN_ID")
       if err ! = nil {
           log.Fatal(err)
       }
       fmt.Println(adminID) // 123456789

       adminPassword, err := env.GetString("ADMIN_PASSWORD")
       if err ! = nil {
           log.Fatal(err)
       }
       fmt.Println(adminPassword) // wtfPassword1234

       adminMode, err := env.GetBool("ADMIN_MODE")
       if err ! = nil {
           log.Fatal(err)
       }
       fmt.Println(adminMode) // true

       adminEmail, err := env.GetString("ADMIN_EMAIL")
       if err ! = nil {
           log.Fatal(err)
       }
       fmt.Println(adminEmail) // admin@localhost
   }
