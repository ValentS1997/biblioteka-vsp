   Далее в том же файле table.go объявим структуру Table, для которой будем объ
являть методы работы с таблицей базы данных. Она достаточно простая, поэтому
каждая база будет содержать только одну таблицу пользователей:

   type Table[T IAttribute] struct {
       head  *Node[T]
       tail  *Node[T]
       Title string
   }

   func NewTable[T IAttribute](title string) *Table[T] {
       return &Table[T]{Title: title}
   }

   func (t *Table[T]) First() T {
       if t.head ! = nil {
           return t.head.Data
