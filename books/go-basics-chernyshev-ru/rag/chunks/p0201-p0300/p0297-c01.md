   func (j *JsonGameStorage) SaveGame(path string, game *m.GameSnapshot) error {
       if !strings.HasSuffix(path, ".json") {
           path += ".json"
       }
       file, err := os.Create(path)
       if err ! = nil {
           return err
       }
       defer file.Close()

       encoder := json.NewEncoder(file)
       err = encoder.Encode(game)
       if err ! = nil {
           return err
       }
       return nil
   }
