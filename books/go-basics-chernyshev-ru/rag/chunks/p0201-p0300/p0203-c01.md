4.4. Обработка паники
Благодаря отложенным функциям и встроенной функции recover у разработчиков
появляется возможность перехватить панику и обработать ее, после чего принять
решение — предотвратить или разрешить экстренное завершение программы:

   // baseURL/part_4/4.4/1.go
   package main

   import (
       "fmt"
       "log"
   )

   type Person struct {
       name string
       age  uint8
   }

   func (p *Person) incAge() {
       p.age+ +
       if p.age > 30 {
           defer func() {
               if err := recover(); err ! = nil { // обработка паники
                   log.Printf("%s has a new life, he (she) is %d years old",
                             p.name, p.age)
               }
           }()
           panic(fmt.Sprintf("%s too old!!!", p.name))
       }
   }
