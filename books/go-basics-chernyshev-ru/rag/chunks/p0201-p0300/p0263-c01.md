   // baseURL/part_5/5.4/go_database/main.go
   package main

   import (
       db "go_database/database"
       "path/filepath"
   )

   func main() {
       db := db.NewDatabase(
           filepath.Join(".", "suai.txt"),
           filepath.Join(".", "unecon.txt"),
       )

       menu := NewMenu(db)
       menu.Loop()
   }
