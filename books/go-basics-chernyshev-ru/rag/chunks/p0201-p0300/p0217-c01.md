216       Глава 5   Работа с файлами и каталогами. Сборка приложения

  После запуска приложения откройте созданный файл, чтобы убедиться, что
в нем есть следующие строчки песни (рис. 5.2).

                                    Рис. 5.2. Запись данных в файл

  Чтобы прочитать данные из файла и вывести их в терминал, можно использовать
функцию Open из пакета os, которая открывает файл в режиме «только для чтения».
Это значит, что записать в него ничего не получится:

   // baseURL/part_5/5.1/7.go
   package main

   import (
       "fmt"
       "io"
       "log"
       "os"
   )

   func main() {
       file, err := os.Open("pirates.txt")
       if err ! = nil {
           log.Fatal(err)
       }
       defer file.Close()

       file.WriteString("jgjgjg") // ничего в файл не добавится!!!
       data := make([]byte, 128)
       for {
           length, err := file.Read(data)
           fmt.Printf("Reading %d byte\n", length)
           if err = = io.EOF { // достигли конца файла?
               break
           }
       }
       fmt.Println(string(data))

       fmt.Println("Text was read from file")
   }
   /*
   Reading 115 byte
   Reading 0 byte
   Пятнадцать человек на сундук мертвеца,
   Йо-хо-хо, и бутылка рома,
   Text was read from file
   */
