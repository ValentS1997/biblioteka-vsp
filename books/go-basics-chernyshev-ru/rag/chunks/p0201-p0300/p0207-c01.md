4.7. Определение типа ошибки
В некоторых ситуациях из функции могут возвращаться различные варианты
ошибок, и их нужно обрабатывать способами, которые отличаются друг от друга.
Для этого принято использовать условную конструкцию swsitch и функцию Is из
пакета errors.
  Для примера добавим в код из предыдущего раздела еще один вариант пользо
вательской ошибки:

   // baseURL/part_4/4.7/1.go
   package main

   import (
       "errors"
       "fmt"
   )

   var ErrDivByZero = errors.New("divide by zero")
   var ErrDivByOne = errors.New("divide by one")

   func div(a, b int) (int, error) {
       if b = = 0 {
           return 0, ErrDivByZero
       }
       if b = = 1 {
           return a, ErrDivByOne
       }
       return a / b, nil
   }

   func printResult(result int, err error) {
       if err ! = nil {
           switch {
           case errors.Is(err, ErrDivByZero):
               fmt.Println(err)
           case errors.Is(err, ErrDivByOne):
               fmt.Printf("result = %d, but was %v\n", result, err)
