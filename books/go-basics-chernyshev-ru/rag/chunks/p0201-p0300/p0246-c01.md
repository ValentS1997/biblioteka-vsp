5.3. Операции с JSON-файлами      245

   func ActorToBytes(actor Actor) []byte {
       actorJson, err := json.MarshalIndent(actor, "", "  ")
       if err ! = nil {
           log.Fatal(err)
       }
       return actorJson
   }

   func main() {
       aboutStr := "Tom Hanks is an actor..."
       // поле FilmsAmount инициализируется значением по умолчанию
       actor := Actor{
           Name:       "Tom Hanks",
           Age:        69,
           AboutActor: &aboutStr,
       }

       actorJson := ActorToBytes(actor)
       // Выводим сериализованные данные
       fmt.Println()
       fmt.Println("First serialized data:")
       fmt.Println(string(actorJson))

       // поля AboutActor и Age инициализируются значениями по умолчанию
       actor = Actor{
           Name:        "Tom Hanks",
           FilmsAmount: 150,
       }

       actorJson = ActorToBytes(actor)
       // Выводим сериализованные данные
       fmt.Println()
       fmt.Println("Second serialized data:")
       fmt.Println(string(actorJson))
   }
   /* First serialized data:
   {
       "name": "Tom Hanks",
       "age": 69,
       "AboutActor": "Tom Hanks is an actor..."
   }

   Second serialized data:
   {
       "name": "Tom Hanks",
       "age": 0,
       "films_amount": 150
   } */

5.3.4. Работа с JSON средствами типа map
В этом подразделе будет приведен код из разряда «можно, но не нужно», так как,
согласно спецификации Go, для работы с форматом JSON лучше использовать
структуры. Это связано с тем, что они предоставляют более безопасный способ рабо
ты с JSON-данными и явно задают типы полей. Но бывают случаи, когда структура
