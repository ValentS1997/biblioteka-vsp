5.2.8. Проверка каталога на существование
Прежде чем создавать очередную директорию или указывать ее в пути к создавае
мому файлу и т. д., необходимо проверить, есть ли она, — это поможет избежать
ошибки времени выполнения, которая экстренно завершит работу приложения.
Для этого можно использовать функцию IsNotExist из пакета os:

   // baseURL/part_5/5.2/12.go
   package main

   import (
       "fmt"
       "os"
   )

   func main() {

       pathToDir := "C:\\Go"

       if _, err := os.Stat(pathToDir); os.IsNotExist(err) {
           // обработка ситуации, когда директории не существует
           fmt.Println("Directory isn't exist")
       } else {
           fmt.Println("Directory already exists")
       }
   }
   // Directory already exists
