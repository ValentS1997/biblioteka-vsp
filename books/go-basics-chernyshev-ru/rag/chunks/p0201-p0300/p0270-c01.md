5.5. Простая база данных по типу «ключ:значение» в формате JSON      269

       "int": 55,
       "map": {
           "a": 1,
           "b": 2
       },
       "str":
       "strList": "-_-"   }         "(づ˝•༝•˝)づ♡",

   Теперь удалим несколько записей:

   // baseURL/part_5/5.5/go_json_store/main.go
   func main() {
       store := jsonstore.NewJSONStore("store.json")

       store.ResetValue("map")
       store.ResetValue("str")

       fmt.Println(store.GetValue("map")) // <nil>
       fmt.Println(store.GetValue("str")) // <nil>
   }

   // baseURL/part_5/5.3/go_json_store/store.json
   {
       "bool": true,
       "double": 99,
       "int": 55,
       "strList": "-_-"
   }
   Когда вы точно знаете, значение какого типа данных хранится по ключу, можете
использовать следующий набор методов JSONStore:
   yy GetBool(key string) bool;
   yy GetInt(key string) int;
   yy GetFloat(key string) float64;
   yy GetString(key string) string;
   yy GetList(key string) []any;
   yy GetMap(key string) map[string]any.
   Значения, возвращаемые последними двумя методами, необходимо переупако
вать в коллекцию нужного типа данных:

   // baseURL/part_5/5.5/go_json_store/main.go
   func main() {
       store := jsonstore.NewJSONStore("store.json")
       store.SetValue("map", map[string]int{"a": 1, "b": 2})
       store.SetValue("strList", []string{"a", "b", "c"})
       store.SetValue("intList", []int{1, 2, 3})

       // Получение и приведение типов
       rawStrList := store.GetList("strList")
       var strList []string
       for _, v := range rawStrList {
           if s, ok := v.(string); ok {
               strList = append(strList, s)
           }
       }
