5.2. Работа с каталогами      233

   Следует отметить, что даже несколько программ или подпрограмм, вызыва
ющих MkdirTemp одновременно, не будут создавать и использовать один и тот же
временный каталог.

5.2.3. Создание цепочки каталогов
Функция MkdirAll(path string, perm FileMode) позволяет создать каталог path
вместе со всеми необходимыми родителями. Биты разрешений perm используются
для всех создаваемых каталогов. Если path уже является каталогом, то функция
ничего не делает и возвращает nil:

   // baseURL/part_5/5.2/5.go
   package main

   import (
       "fmt"
       "log"
       "os"
       "path/filepath"
   )

   const tempFileName = "myTempFile.dat"

   func createdFile(path string) {
       filepath := filepath.Join(path, tempFileName)
       myFile, err := os.Create(filepath)
       if err ! = nil {
           log.Fatal(err)
       }
       defer myFile.Close()
       fmt.Println("File created with path: ", filepath)
   }

   func main() {
       path := filepath.Join("firstDir", "secondDir", "new")
       os.MkdirAll(path, 0777)
       createdFile(path)
   }
   // File created with path:  firstDir\secondDir\new\myTempFile.dat
