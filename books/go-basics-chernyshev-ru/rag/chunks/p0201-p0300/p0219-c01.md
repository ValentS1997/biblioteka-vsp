  Для чтения всего файла полностью за один раз необходимо использовать функ
цию ReadAll из пакета io:

   // baseURL/part_5/5.1/9.go
   package main

   import (
       "fmt"
       "io "
       "log"
       "os"
   )

   func main() {
       file, err := os.Open("pirates.txt")
       if err ! = nil {
           log.Fatal(err)
       }
       defer file.Close()

       data, err := io.ReadAll(file)
       if err ! = nil {
           log.Fatal(err)
       }
       fmt.Println(string(data))
   }
   // Пятнадцать человек на сундук мертвеца,
   // Йо-хо-хо, и бутылка рома,
