# Диаграмма — работа со списками

~~~mermaid
flowchart TD
    A["Есть список"] --> B{"Что нужно сделать?"}
    B -- "Прочитать" --> C["items[index] / items[-1]"]
    B -- "Изменить" --> D["items[index] = value"]
    B -- "Добавить" --> E{"Куда?"}
    E -- "В конец" --> E1["append(value)"]
    E -- "В позицию" --> E2["insert(index, value)"]
    B -- "Удалить" --> F{"Нужно использовать удалённое значение?"}
    F -- "Нет, индекс известен" --> F1["del items[index]"]
    F -- "Да" --> F2["pop() / pop(index)"]
    F -- "Известно только значение" --> F3["remove(value)"]
    B -- "Упорядочить" --> G{"Менять исходный список?"}
    G -- "Нет" --> G1["sorted(items)"]
    G -- "Да: сортировка" --> G2["items.sort()"]
    G -- "Да: обратить текущий порядок" --> G3["items.reverse()"]
    B -- "Посчитать" --> H["len(items)"]

    C --> V["Проверить результат"]
    D --> V
    E1 --> V
    E2 --> V
    F1 --> V
    F2 --> V
    F3 --> V
    G1 --> V
    G2 --> V
    G3 --> V
    H --> V
~~~

## Диагностика IndexError

~~~mermaid
flowchart TD
    A["IndexError: list index out of range"] --> B["Прочитать traceback"]
    B --> C["Проверить запрошенный индекс"]
    C --> D{"Off-by-one?"}
    D -- "Да" --> E["Уменьшить индекс на 1"]
    D -- "Неясно" --> F["Вывести список"]
    F --> G["Проверить len(list)"]
    G --> H["Сопоставить доступные индексы"]
    E --> I["Повторить запуск"]
    H --> I
    I --> J{"Ошибка устранена?"}
    J -- "Да" --> K["Оставить проверенное изменение"]
    J -- "Нет" --> B
~~~
