Инструменты Docker Compose и Podman Compose
Docker Compose — это инструмент для определения и запуска многоконтей-
нерных приложений. Он использует YAML-файл для настройки необходимых
приложению сервисов, сетей и томов, позволяя разработчикам определить всю
программную инфраструктуру в одном месте. Инструменты Docker Compose
и Podman Compose упрощают оркестрацию сложных приложений, помогая
управлять множеством контейнеров как единым стеком.
Следующий файл, compose.yaml, создает два контейнера для стека приложений
WordPress с помощью одной команды docker compose или podman compose:

# compose.yaml
services:
  db:
    image: docker.io/library/mariadb
    command: '--default-authentication-plugin=mysql_native_password'
    volumes:
      - db_data:/var/lib/mysql
    restart: always
    environment:
      - MYSQL_ROOT_PASSWORD=somewordpress
      - MYSQL_DATABASE=wordpress
      - MYSQL_USER=wordpress
      - MYSQL_PASSWORD=wordpress
    expose:
      - "3306"
      - "33060"
    networks:
      - wordpress
  wordpress:
    image: wordpress:latest
    ports:
      - "8081:80"
    restart: always
    environment:
      - WORDPRESS_DB_HOST=db
      - WORDPRESS_DB_USER=wordpress
      - WORDPRESS_DB_PASSWORD=wordpress
      - WORDPRESS_DB_NAME=wordpress
    networks:
      - wordpress
volumes:
  db_data:
networks:
  wordpress: {}

netes способна оркестровать все эти операции с контейнерами.
